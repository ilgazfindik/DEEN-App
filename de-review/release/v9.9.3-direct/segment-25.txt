(()=>{
  const VERSION='10.4.2-FILL-BALANCE',BUILD='9142fillbalance1';
  function mark(){

    if(!frame?.classList.contains('v1040-fill-interactive'))return false;
    frame.dataset.deenFillBalance='v1042';
    document.documentElement.dataset.deenFillBalance='v1042';

  document.addEventListener('deen:question-graded',()=>requestAnimationFrame(mark));
  requestAnimationFrame(mark);
  window.DEEN_FILL_BALANCE={
    version:VERSION,
    build:BUILD,
    apply:mark,
    snapshot:()=>({
      active:!!document.querySelector('#lesson .question-frame[data-deen-fill-balance="v1042"]'),
      renderer:document.querySelector('#lesson .question-frame')?.dataset?.deenFillRenderer||null
    })
  };
})();
