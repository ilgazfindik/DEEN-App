<script id="deen-v10946-fatiha-inline-audio-js">
(()=>{
if(window.__deenV10946FatihaInlineAudio)return;window.__deenV10946FatihaInlineAudio=true;

function q(){return DEEN_QURAN?.ensureState?.()||null}
function ayah(){return Math.max(1,Math.min(7,Number(q()?.profile?.currentAyah)||1))}
function isFatiha(){return q()?.profile?.currentSurah==='fatiha'}
function key(){return 'fatiha:'+ayah()}

function audioReady(){
  try{return !!DEEN_QURAN_AUDIO?.verifiedAudio?.('fatiha',ayah())}catch(_){return false}
}
function syncButton(){
  const b=document.querySelector?.('.deen-fatiha-inline-speaker');
  if(!b)return false;
  const s=DEEN_QURAN_AUDIO?.state?.()||{};
  const playing=s.status==='playing'&&s.playingKey===key();
  b.classList.toggle('playing',playing);
  b.textContent=playing?'Ⅱ':'🔊';
  b.setAttribute('aria-label',playing?'Ayet sesini duraklat':'Ayet sesini dinle');
  return true;
}
async function toggle(){
  const n=ayah();
  if(!audioReady()){
    const h=document.querySelector?.('.deen-fatiha-inline-hint');
    if(h)h.textContent='Bu ayetin sesi henüz hazır değil.';
    return false;
  }
  const out=await DEEN_QURAN_AUDIO.toggle('fatiha',n);
  syncButton();
  return out;
}
function decorate(){
  const root=document.querySelector?.('.deen-qmulti-session');
  if(!root||!isFatiha())return false;

  root.setAttribute('data-fatiha-inline-audio','1');
  const card=root.querySelector('.deen-ikhlas-card');
  if(!card)return false;

  let button=card.querySelector('.deen-fatiha-inline-speaker');
  if(!button){
    const h2=card.querySelector('h2');
    if(h2){
      const row=document.createElement('div');
      row.className='deen-fatiha-inline-row';
      h2.parentNode.insertBefore(row,h2);
      row.appendChild(h2);
      button=document.createElement('button');
      button.type='button';
      button.className='deen-fatiha-inline-speaker';
      button.setAttribute('onclick','DEEN_QURAN_FATIHA_INLINE_AUDIO.toggle()');
      row.appendChild(button);
    }
  }

  let hint=card.querySelector('.deen-fatiha-inline-hint');
  if(!hint){
    hint=document.createElement('p');
    hint.className='deen-fatiha-inline-hint';
    hint.textContent='Hoparlöre dokun: bu ayetin ElevenLabs kaydını dinle.';
    const row=card.querySelector('.deen-fatiha-inline-row');
    if(row)row.insertAdjacentElement('afterend',hint);
  }
  syncButton();
  return true;
}
function audit(){
  const rows=[];
  for(let n=1;n<=7;n++){
    const a=DEEN_QURAN_AUDIO?.verifiedAudio?.('fatiha',n);
    rows.push({ayah:n,ready:!!a,url:a?.url||null});
  }
  return{
    ok:rows.every(x=>x.ready),
    wikimediaKept:!!window.DEEN_QURAN_CHAPTER_AUDIO?.source?.('fatiha'),
    inlineFiles:rows
  };
}

let timer=null;
function schedule(){clearTimeout(timer);timer=setTimeout(decorate,35)}
document.addEventListener?.('click',e=>{if(e.target?.closest?.('#v5Sheet'))schedule()},true);
document.addEventListener?.('deen:navigated',schedule);
if(typeof MutationObserver!=='undefined')new MutationObserver(schedule).observe(document.body,{childList:true,subtree:true});
setInterval(syncButton,300);
schedule();

window.DEEN_QURAN_FATIHA_INLINE_AUDIO={toggle,decorate,audit,syncButton,ayah};
document.documentElement.dataset.deenFatihaInlineAudio='v10946';
})();
</script>
