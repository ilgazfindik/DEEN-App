<script id="deen-v10930-quran-final-qa-js">
(()=>{
  if(window.__deenV10930QuranFinalQa)return;
  window.__deenV10930QuranFinalQa=true;

  const IDS=['fatiha','ikhlas','falaq','nas','kafirun','nasr','masad','fil','quraysh','maun','sharh','tin','bayyina','kawthar','asr'];
  const COUNTS={fatiha:7,ikhlas:4,falaq:5,nas:6,kafirun:6,nasr:3,masad:5,fil:5,quraysh:4,maun:7,sharh:8,tin:8,bayyina:8,kawthar:3,asr:3};
  const qstate=()=>DEEN_QURAN?.ensureState?.()||state?.quranJourney||null;
  const validDate=v=>typeof v==='string'&&Number.isFinite(Date.parse(v));

  function repairReviewQueue(){
    const q=qstate();
    if(!q)return {changed:false,before:0,after:0};
    q.review=q.review&&typeof q.review==='object'?q.review:{queue:[],legacyQueue:[],history:[]};
    q.review.queue=Array.isArray(q.review.queue)?q.review.queue:[];
    q.review.legacyQueue=Array.isArray(q.review.legacyQueue)?q.review.legacyQueue:[];

    const before=q.review.queue.length;
    const best=new Map();
    const invalid=[];
    for(const item of q.review.queue){
      const sid=String(item?.surahId||'');
      const ayah=Number(item?.ayah);
      if(!IDS.includes(sid)||!Number.isInteger(ayah)||ayah<1||ayah>COUNTS[sid]||!validDate(item?.dueAt)){
        invalid.push(item);
        continue;
      }
      const k=sid+':'+ayah;
      const prev=best.get(k);
      if(!prev||Date.parse(item.dueAt)<Date.parse(prev.dueAt))best.set(k,item);
    }
    q.review.queue=[...best.values()].sort((a,b)=>Date.parse(a.dueAt)-Date.parse(b.dueAt));
    if(invalid.length)q.review.legacyQueue.push(...invalid);
    const changed=before!==q.review.queue.length||invalid.length>0;
    if(changed)DEEN_QURAN?.persistState?.();
    return {changed,before,after:q.review.queue.length,invalid:invalid.length};
  }

  function audit(){
    const q=qstate();
    const checks={};
    checks.schema=!!q&&Number(q.schemaVersion)===2&&Number(q.version)===2;
    checks.currentSurah=!!q&&IDS.includes(q.profile?.currentSurah||q.currentSurah||'fatiha');
    const sid=q?.profile?.currentSurah||q?.currentSurah||'fatiha';
    const ayah=Number(q?.profile?.currentAyah||1);
    checks.currentAyah=IDS.includes(sid)&&Number.isInteger(ayah)&&ayah>=1&&ayah<=COUNTS[sid];

    const content={};
    for(const id of IDS){
      let ready=0;
      try{ready=DEEN_QURAN_CONTENT?.getSurah?.(id)?.length||0}catch(_){}
      content[id]={ready,total:COUNTS[id],ok:ready===COUNTS[id]};
    }
    checks.contentReady=Object.values(content).every(x=>x.ok);

    const queue=Array.isArray(q?.review?.queue)?q.review.queue:[];
    const keys=queue.map(x=>String(x?.surahId)+':'+Number(x?.ayah));
    checks.reviewQueueUnique=new Set(keys).size===keys.length;
    checks.reviewDates=queue.every(x=>validDate(x?.dueAt));

    let audioSafe=true;
    try{
      const rows=DEEN_QURAN_AUDIO?.audit?.()||[];
      audioSafe=rows.every(x=>!x.playable||(x.rights_status==='cleared'&&!!x.license&&!!x.rights_reference));
    }catch(_){audioSafe=false}
    checks.audioRightsGate=audioSafe;

    checks.sheet=typeof document==='undefined'||!!document.getElementById?.('v5Sheet');
    checks.quranRoute=typeof document==='undefined'||!!document.getElementById?.('leagueScreen');
    checks.rtlCss=typeof document==='undefined'||!!document.getElementById?.('deen-v10930-quran-final-qa-css');

    const failures=Object.entries(checks).filter(([,ok])=>!ok).map(([k])=>k);
    return {
      ok:failures.length===0,
      failures,
      checks,
      content,
      queueCount:queue.length,
      schemaVersion:q?.schemaVersion||null,
      current:{surahId:sid,ayah}
    };
  }

  function stopAudio(){
    try{DEEN_QURAN_AUDIO?.stop?.(false)}catch(_){}
  }

  const repair=repairReviewQueue();
  const result=audit();
  document.documentElement.dataset.deenQuranQa=result.ok?'pass':'fail';

  document.addEventListener?.('close',e=>{
    if(e.target?.id==='v5Sheet')stopAudio();
  },true);
  document.addEventListener?.('deen:navigating',stopAudio);
  document.addEventListener?.('deen:navigated',()=>{
    const r=audit();
    document.documentElement.dataset.deenQuranQa=r.ok?'pass':'fail';
  });

  window.DEEN_QURAN_QA={
    audit,
    repairReviewQueue,
    stopAudio,
    initial:{repair,result},
    version:'10.9.30'
  };
})();
</script>
