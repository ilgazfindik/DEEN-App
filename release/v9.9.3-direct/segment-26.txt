<script id="deen-v1043-fill-feedback-handoff-js">
(()=>{
  if(window.__deenV1043FillFeedbackHandoff)return;
  window.__deenV1043FillFeedbackHandoff=true;

  const VERSION='10.4.3-FILL-FEEDBACK-HANDOFF';
  const BUILD='9143fillfeedback1';
  let handoffs=0;
  let resets=0;
  let last=null;

  const frame=()=>document.querySelector('#lesson .question-frame');

  function isFill(detail){
    const q=detail?.question;
    const type=String(q?.activity_type||q?.question_type||q?.type||'').toLowerCase();
    const f=frame();
    return type==='fill_blank' || !!f?.classList.contains('v1040-fill-interactive');
  }

  function reset(){
    const f=frame();
    f?.classList.remove('v1043-fill-graded');
    document.querySelectorAll('.v1040-drag-ghost').forEach(el=>el.remove());
    const btn=document.getElementById('continueBtn');
    if(btn)delete btn.dataset.v1043Ready;
    resets++;
    return true;
  }

  function handoff(detail){
    if(!isFill(detail))return false;
    const f=frame();
    const fb=document.getElementById('feedback');
    const btn=document.getElementById('continueBtn');
    if(!f||!fb||!btn)return false;

    f.classList.add('v1043-fill-graded');

    /*
      completeQuestion has already opened #feedback before the authoritative
      question-graded event is dispatched. Apply the existing compact feedback
      layout immediately instead of waiting for its legacy 330 ms delay.
    */
    try{window.DEEN_FEEDBACK_LAYOUT?.apply?.(detail?.correct===true,detail?.question)}catch(_){}

    btn.textContent='Devam et';
    btn.disabled=false;
    btn.dataset.v1043Ready='1';

    document.querySelectorAll('.v1040-drag-ghost').forEach(el=>el.remove());

    handoffs++;
    last={
      at:Date.now(),
      epoch:Number(detail?.epoch)||0,
      key:String(detail?.key||''),
      correct:detail?.correct===true,
      feedbackVisible:fb.classList.contains('show'),
      answerHidden:getComputedStyle(document.getElementById('answerArea')).display==='none'
    };

    try{document.dispatchEvent(new CustomEvent('deen:fill-feedback-ready',{detail:{...last,version:VERSION,build:BUILD}}))}catch(_){}
    return true;
  }

  document.addEventListener('deen:question-rendering',reset);
  document.addEventListener('deen:question-graded',event=>handoff(event?.detail||{}));
  document.addEventListener('deen:lesson-result',reset);

  window.DEEN_FILL_FEEDBACK_HANDOFF={
    version:VERSION,
    build:BUILD,
    apply:handoff,
    reset,
    snapshot:()=>({
      version:VERSION,
      build:BUILD,
      active:!!frame()?.classList.contains('v1043-fill-graded'),
      renderer:frame()?.dataset?.deenFillRenderer||null,
      feedbackVisible:!!document.getElementById('feedback')?.classList.contains('show'),
      answerVisible:(()=>{
        const el=document.getElementById('answerArea');
        return !!el&&getComputedStyle(el).display!=='none';
      })(),
      continueReady:document.getElementById('continueBtn')?.dataset?.v1043Ready==='1',
      handoffs,
      resets,
      last:last?{...last}:null
    })
  };
})();
</script>
