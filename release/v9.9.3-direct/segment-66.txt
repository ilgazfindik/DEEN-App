 const sameArray=(a,b)=>{
  if(!Array.isArray(a))return !Array.isArray(b);
  return Array.isArray(b)&&a.length===b.length&&a.every((x,i)=>x===b[i]);
 };
 const samePairs=(a,b)=>{
  if(!Array.isArray(a))return !Array.isArray(b)||b.length===0;
  return Array.isArray(b)&&a.length===b.length&&a.every((x,i)=>x?.left===b[i]?.left&&x?.right===b[i]?.right);
 };
 const blankCount=s=>(String(s||'').match(/___/g)||[]).length;

 function userVisibleStrings(t){
  const out=[t?.question_ar,t?.correct_answer_ar,t?.explanation_ar];
  (t?.options_ar||[]).forEach(x=>out.push(x));
  (t?.pairs_ar||[]).forEach(p=>out.push(p?.left,p?.right));
  (t?.items_ar||[]).forEach(x=>out.push(x));
  (t?.correct_order_ar||[]).forEach(x=>out.push(x));
  return out.filter(x=>typeof x==='string'&&x.trim());
 }

 function validArabicEntry(t){
  if(!t||t.expert_approved!==false)return false;
  if(typeof t.question_ar!=='string'||!t.question_ar.trim())return false;
  if(typeof t.explanation_ar!=='string'||!t.explanation_ar.trim())return false;
  const values=userVisibleStrings(t);
  if(values.some(v=>/[A-Za-zÇĞİÖŞÜçğıöşü]/.test(v.replace(/\bDEEN\b/g,''))))return false;
  if(t.teaching_card_ar&&Object.values(t.teaching_card_ar).some(v=>typeof v!=='string'||!v.trim()||/[A-Za-zÇĞİÖŞÜçğıöşü]/.test(v)))return false;
  return /[\u0600-\u06FF]/.test(t.question_ar)&&/[\u0600-\u06FF]/.test(t.explanation_ar);
 }

 function asDisplayQuestion(base,translated,isRuntimeVariant=false){
  if(!base||!validArabicEntry(translated))return null;
  if(base.question_tr!==translated.source_question_tr)return null;
  if(base.activity_type!==translated.activity_type)return null;
  if(blankCount(base.question_tr)!==blankCount(translated.question_ar))return null;

  if(isRuntimeVariant){
   if(base.activity_type==='matching'){
    if(!samePairs(base.pairs,translated.source_pairs_tr))return null;
   }else{
    if(!sameArray(base.options_tr,translated.source_options_tr))return null;
    if(!samePairs(base.pairs,translated.source_pairs_tr))return null;
   }
   if(!sameArray(base.items,translated.source_items_tr))return null;
   if(!sameArray(base.correct_order,translated.source_correct_order_tr))return null;
  }

  if(base.activity_type!=='matching' && base.activity_type!=='build_sequence' &&
     base.correct_answer!==translated.source_correct_answer_tr)return null;

  if(base.activity_type==='matching'){
   if(!Array.isArray(base.pairs)||!Array.isArray(translated.pairs_ar)||translated.pairs_ar.length!==base.pairs.length)return null;
   if(translated.pairs_ar.some(p=>typeof p?.left!=='string'||typeof p?.right!=='string'||!p.left.trim()||!p.right.trim()))return null;
  }else if(Array.isArray(base.options_tr)){
   if(!Array.isArray(translated.options_ar)||translated.options_ar.length!==base.options_tr.length)return null;
   const correctIndex=base.options_tr.indexOf(base.correct_answer);
   if(correctIndex<0||translated.options_ar[correctIndex]!==translated.correct_answer_ar)return null;
   if(translated.options_ar.filter(x=>x===translated.correct_answer_ar).length!==1)return null;
  }else if(base.activity_type==='build_sequence'||Array.isArray(base.correct_order)){
   if(!Array.isArray(base.items)||!Array.isArray(translated.items_ar)||translated.items_ar.length!==base.items.length)return null;
   if(!Array.isArray(base.correct_order)||!Array.isArray(translated.correct_order_ar)||translated.correct_order_ar.length!==base.correct_order.length)return null;
  }else if(!translated.correct_answer_ar){
   return null;
  }

  return {
   ...base,
   question_tr:translated.question_ar,
   options_tr:Array.isArray(translated.options_ar)?[...translated.options_ar]:null,
   correct_answer:translated.correct_answer_ar||null,
   pairs:Array.isArray(translated.pairs_ar)?translated.pairs_ar.map(p=>({...p})):base.pairs,
   items:Array.isArray(translated.items_ar)?[...translated.items_ar]:base.items,
   correct_order:Array.isArray(translated.correct_order_ar)?[...translated.correct_order_ar]:base.correct_order,
   explanation_tr:translated.explanation_ar,
   stage_title:typeof arabizeDynamicText==='function'?arabizeDynamicText(base.stage_title||''):base.stage_title,
   topic:arabicU02ConceptNames[base.concept_id]||(typeof arabizeDynamicText==='function'?arabizeDynamicText(base.topic||''):base.topic),
   teaching_card:translated.teaching_card_ar?{...translated.teaching_card_ar}:base.teaching_card,
   __deenArabicPilot:true,
   __deenArabicFullBank:true,
   __deenSourceQuestion:base,
   __deenRuntimeVariant:!!isRuntimeVariant
  };
 }

 pilot.translateQuestion=function(base){
  if(!pilot.ready||!base)return null;
  const live=pilot.live_translations[base.id];
  const variant=pilot.runtime_variants[base.id];
  const translated=live||variant||pilot.translations[base.id];
  return asDisplayQuestion(base,translated,!!live||!!variant);
 };

 pilot.canStart=id=>pilot.ready&&supportedStages.has(id);

 pilot.questionsFor=function(bank,stageId,count,shuffleFn){
  if(!pilot.ready||!Array.isArray(bank)||!supportedStages.has(stageId))return [];
  const sourcePool=bank.filter(q=>q.stage_id===stageId);
  if(!sourcePool.length)return [];
  const pool=sourcePool.map(q=>pilot.translateQuestion(q)).filter(Boolean);
  if(pool.length!==sourcePool.length)return [];
  const random=typeof shuffleFn==='function'?shuffleFn:(list=>[...list]);
  const target=stageId.endsWith('S07')?10:(Number.isFinite(Number(count))?Number(count):7);
  const groups={};
  pool.forEach(q=>(groups[q.activity_type]??=[]).push(q));
  let chosen=[];
  for(const key of Object.keys(groups)){
   const g=random(groups[key]);
   if(g[0])chosen.push(g[0]);
  }
  chosen=random(chosen).slice(0,target);
  const seen=new Set(chosen.map(q=>q.id));
  for(const q of random(pool)){
   if(chosen.length>=target)break;
   if(!seen.has(q.id)){chosen.push(q);seen.add(q.id)}
  }
  return random(chosen).slice(0,target);
 };

 async function preload(){
  pilot.loading=true;pilot.failure='';pilot.ready=false;

   }
   const [units,draft,liveFile]=await Promise.all([
    Promise.all(unitJobs),
    fetchJson('./assets/i18n/questions.ar.draft.json?v='+token),
    fetchJson('./assets/i18n/questions/ar/runtime/U01.live.json?v='+token)
   ]);
   const translations=Object.create(null);
   for(const file of units){
    if(file?.locale!=='ar'||file?.release_ready!==false||file?.verified_by_religious_expert!==false)throw Error('Unsafe Arabic unit metadata');
    Object.assign(translations,file.translations||{});
   }
   if(Object.keys(translations).length!==1500)throw Error('Arabic base bank count '+Object.keys(translations).length);
   if(Object.values(translations).some(x=>!validArabicEntry(x)))throw Error('Invalid Arabic base entry');
   const variants=draft?.runtime_variants||{};
   if(Object.keys(variants).length!==20)throw Error('Arabic runtime variant count '+Object.keys(variants).length);
   if(Object.values(variants).some(x=>!validArabicEntry(x)))throw Error('Invalid Arabic runtime variant');
   if(liveFile?.locale!=='ar'||liveFile?.release_ready!==false||liveFile?.verified_by_religious_expert!==false)throw Error('Unsafe Arabic live metadata');
   const live=liveFile?.translations||{};
   if(Object.keys(live).length!==84)throw Error('Arabic live U01 count '+Object.keys(live).length);
   if(Object.values(live).some(x=>!validArabicEntry(x)))throw Error('Invalid Arabic live U01 entry');
   pilot.translations=translations;
   pilot.runtime_variants=variants;
   pilot.live_translations=live;
   pilot.ready=true;
  }catch(e){
   pilot.failure=String(e?.message||e);
   pilot.ready=false;
   console.error('[DEEN Arabic full bank]',e);
  }finally{
   pilot.loading=false;
  }
  return pilot.ready;
 }

 function localizeChrome(root=document.getElementById('lesson')||document.body){
  if(locale()!=='ar'||!root)return;

  // Unit titles are keyed by unit number so Turkish source names can never leak into Arabic UI.
  for(const [n,name] of Object.entries(arabicUnitNames)){
   document.querySelectorAll('#heroU'+n+' .hero-copy h1,#v5Unit'+n+' .hero-copy h1,[data-unit="U'+String(n).padStart(2,'0')+'"] .hero-copy h1').forEach(el=>{
    if(el.textContent!==name)el.textContent=name;
   });
  }
  root.querySelectorAll?.('.v1017-title').forEach(el=>{
   if(!/[\u0600-\u06FF]/.test(el.textContent||''))el.innerHTML='اعثر على <span>الأزواج المتطابقة</span>';
  });
  root.querySelectorAll?.('.v1017-subtitle').forEach(el=>el.textContent='صِل المفاهيم بمعانيها الصحيحة.');
  const walker=document.createTreeWalker(root,NodeFilter.SHOW_TEXT);
  const nodes=[];let n;
  while((n=walker.nextNode()))nodes.push(n);
  for(const node of nodes){
   const raw=node.nodeValue||'',trim=raw.trim();
   let out=chromeMap[trim];
   if(!out){
    const m=trim.match(/^(\d+)\s*\/\s*(\d+)\s*eşleşme$/i);
    if(m)out=m[1]+' / '+m[2]+' مطابقات';
   }
   if(!out&&trim.startsWith('Öğrenme döngüsü · ')){
    const phase=trim.slice('Öğrenme döngüsü · '.length);
    const phaseAr=chromeMap[phase]||phase;
    out='مسار التعلّم · '+phaseAr;
   }
   if(!out){
    const dynamic=arabizeDynamicText(trim);
    if(dynamic!==trim)out=dynamic;
   }
   if(out)node.nodeValue=raw.replace(trim,out);
  }
  root.querySelectorAll?.('[aria-label],[title],[placeholder]').forEach(el=>{
   for(const attr of ['aria-label','title','placeholder']){
    const raw=el.getAttribute?.(attr);
    if(!raw)continue;
    const converted=arabizeDynamicText(raw);
    if(converted!==raw)el.setAttribute(attr,converted);
   }
  });
  localizeLearningLoop(root);

  const rv=document.getElementById('resultView');
  if(rv){
   rv.setAttribute('dir','rtl');
   rv.querySelectorAll('*').forEach(el=>{
    if(el.children.length)return;
    const raw=el.textContent||'',trim=raw.trim();
    if(!trim)return;
    const converted=arabizeDynamicText(trim);
    if(converted!==trim)el.textContent=raw.replace(trim,converted);
   });

   const conceptLine=rv.querySelector('#v5ResultExtra > p:not(.v5-muted)');
   if(conceptLine){
    const t=(conceptLine.textContent||'').trim();
    if(/[ÇĞİÖŞÜçğıöşü]/.test(t)||/\b(İslam|Temel|kavram|kimliği|namaz|oruç|iman|ibadet|ahlak|peygamber|vahiy|kader)\b/i.test(t)){
     conceptLine.textContent='مفاهيم الدرس · مراجعة وتطبيق';
    }
   }
  }
 }

 };
