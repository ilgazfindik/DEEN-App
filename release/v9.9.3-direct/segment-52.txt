function exactWords(c,n,spanCount){
  const row=exactTimingEntries[c.id+':'+Number(n)];
  const words=Array.isArray(row?.words)?row.words:[];

  if(clean.some(w=>!Number.isFinite(w.start)||!Number.isFinite(w.end)||w.end<w.start))return null;
  return clean;
}

function syncText(){
  const c=activeSurah?cfg(activeSurah):currentCfg();
  if(!c||!player||currentAyah!==visibleAyah(c))return false;
  const ar=prepareSyncedText(c);
  if(!ar)return false;

  const spans=[...ar.querySelectorAll('.deen-ayah-sync-word')];
  const bar=ar.querySelector('.deen-ayah-sync-track i');
  const dur=Number(player.duration),t=Number(player.currentTime)||0;
  if(!Number.isFinite(dur)||dur<=0||!spans.length)return false;

  const progress=Math.max(0,Math.min(1,t/dur));
  if(bar)bar.style.width=(progress*100).toFixed(2)+'%';

  if(status==='ended'){
    spans.forEach(w=>{w.classList.add('past');w.classList.remove('current')});
    return true;
  }

  const exact=exactWords(c,currentAyah,spans.length);
  if(exact){
    let active=-1;
    for(let i=0;i<exact.length;i++){
      if(t>=exact[i].start&&t<exact[i].end){active=i;break}
    }
    spans.forEach((w,i)=>{
      w.classList.toggle('past',t>=exact[i].end);
      w.classList.toggle('current',i===active);
    });
    ar.dataset.syncMode='exact';
    return true;
  }

  const lead=Math.min(c.leadMax,dur*c.leadFactor);
  const tail=Math.min(c.tailMax,dur*c.tailFactor);
  const usable=Math.max(.1,dur-lead-tail);
  const p=Math.max(0,Math.min(.999,(t-lead)/usable));
  const weights=spans.map(w=>Math.max(1,Number(w.dataset.syncWeight)||1));
  const total=weights.reduce((a,b)=>a+b,0);
  let acc=0,idx=0;
  for(let i=0;i<weights.length;i++){
    acc+=weights[i]/total;
    if(p<acc){idx=i;break}
  }
  spans.forEach((w,i)=>{
    w.classList.toggle('past',i<idx);
    w.classList.toggle('current',i===idx&&t>=lead);
  });
  ar.dataset.syncMode='weighted';
  return true;
}

    const previous=activeSurah?cfg(activeSurah):null;
    try{p.pause()}catch(_){}
    if(previous)resetSyncedText(true,previous);
    activeSurah=c.id;
    currentAyah=n;
    resetSyncedText(true,c);
    prepareSyncedText(c);
    p.src=audioUrl(c,n)||'';
    try{p.currentTime=0}catch(_){}
  }else if(status==='ended'){
    try{p.currentTime=0}catch(_){}
    resetSyncedText(true,c);
  }

  hint('Ayet '+n+' dinleniyor…',false,c);
  try{
    const out=p.play();
    if(out&&typeof out.then==='function')await out;
    status='playing';syncButton(c);startRaf();syncText();
    DEEN_QURAN_ANALYTICS?.track?.('audio_play',{surahId:c.id,mode:'ayah_local_universal'});
    return true;
  }catch(e){

  }

    h2.parentNode.insertBefore(row,h2);
    row.appendChild(h2);
  }

    button.type='button';

  }

    button.dataset.deenAudioBound='1';
    button.addEventListener('click',()=>{
      const id=button.dataset.surahAudio||currentId();
      return toggle(id);
    });
  }

  let h=card.querySelector('.deen-ayah-inline-hint');
  if(!h){
    h=document.createElement('p');
    h.className='deen-ayah-inline-hint';
    row?.insertAdjacentElement('afterend',h);
  }
  if(h&&!h.classList.contains('error')){const text='Hoparlöre dokun: Ayet '+visibleAyah(c,root)+' kaydını dinle.';const value=window.DEEN_ARABIC_UI?.active?.()?DEEN_ARABIC_UI.translate(text):text;if(h.textContent!==value)h.textContent=value;}

  const n=visibleAyah(c,root);
  if(activeSurah&&activeSurah!==c.id)stop();
  if(activeSurah===c.id&&currentAyah!==null&&currentAyah!==n)stop();

  prepareSyncedText(c);
  syncText();
  syncButton(c);
  return true;
}

              rights_reference:'Generated/provided by DEEN project owner for in-app use'
            },
            verification_status:'production_ready',
            rights_status:'cleared'
          },
          updated_at:new Date().toISOString()
        });
        bound++;
      }catch(_){missing++}
    }
  }
  return {bound,missing,total:TOTAL_AYAHS};
}
function filesAudit(){
  return IDS.flatMap(id=>CONFIG[id].files.map((file,i)=>({
    surahId:id,ayah:i+1,file,url:CONFIG[id].base+file
  })));
}
function regressionAudit(){
  const files=filesAudit();
  const duplicateUrls=files.filter((x,i,a)=>a.findIndex(y=>y.url===x.url)!==i).map(x=>x.url);
  const configIssues=IDS.flatMap(id=>{
    const c=CONFIG[id],issues=[];
    if(c.id!==id)issues.push(id+':id_mismatch');
    if(c.files.length!==c.count)issues.push(id+':file_count');
    if(!c.root)issues.push(id+':root_missing');
    return issues;
  });
  const registryRows=files.map(x=>({
    ...x,
    contentReady:!!DEEN_QURAN_CONTENT?.internalGet?.(x.surahId,x.ayah),
    verifiedAudio:!!DEEN_QURAN_AUDIO?.verifiedAudio?.(x.surahId,x.ayah)
  }));
  const failedRegistry=registryRows.filter(x=>!x.contentReady||!x.verifiedAudio);
  const ok=IDS.length===16&&TOTAL_AYAHS===90&&files.length===90&&
    !duplicateUrls.length&&!configIssues.length&&!failedRegistry.length;
  document.documentElement.dataset.deenQuranAudioQa=ok?'pass':'fail';
  return {
    ok,version:'v10962',build:BUILD,
    surahs:IDS.length,ayahs:TOTAL_AYAHS,files:files.length,
    duplicateUrls,configIssues,
    registryReady:registryRows.length-failedRegistry.length,
    registryTotal:registryRows.length,
    failedRegistry,
    exactTimingAyahs:Object.keys(exactTimingEntries).filter(k=>Array.isArray(exactTimingEntries[k]?.words)&&exactTimingEntries[k].words.length).length,
    timingStatus,
    timingLoadedAt,
    singlePlayer:true,
    singleObserver:true,
    dynamicSharedButton:true,
    wikimediaKept:!!window.DEEN_QURAN_CHAPTER_AUDIO
  };
}
function audit(){
  const qa=regressionAudit();
  const files=filesAudit();
  return {
    ...qa,
    version:'v10962',
    build:BUILD,
    surahs:IDS.length,
    ayahs:TOTAL_AYAHS,
    files:files.length,
    currentSurah:currentId(),
    wikimediaKept:!!window.DEEN_QURAN_CHAPTER_AUDIO,
    singlePlayer:true,
    singleObserver:true,
    configs:JSON.parse(JSON.stringify(CONFIG))
  };
}
