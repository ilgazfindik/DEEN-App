<script id="deen-v10931-quran-device-audit-js">
(()=>{
  if(window.__deenV10931QuranDeviceAudit)return;
  window.__deenV10931QuranDeviceAudit=true;

  const ROOT_SELECTORS=[
    '.deen-quran-library-wrap','.deen-ikhlas-session','.deen-mem-session',
    '.deen-practice-session','.deen-srs-session','.deen-dailyverse',
    '.deen-wbw','.deen-qprog'
  ];
  const ARABIC_SELECTORS=[
    '.deen-ikhlas-arabic','.deen-mem-arabic','.deen-practice-arabic',
    '.deen-srs-arabic','.deen-dailyverse-arabic','.deen-wbw-verse','.deen-wbw-ar'
  ];
  const BUTTON_SELECTORS=[
    '.deen-ikhlas-btn','.deen-mem-btn','.deen-practice-btn','.deen-srs-btn',
    '.deen-dailyverse-option','.deen-wbw-btn','.deen-practice-chip','.deen-wbw-word'
  ];

  function visible(el){
    if(!el||typeof el.getBoundingClientRect!=='function')return false;
    const r=el.getBoundingClientRect();
    return r.width>0&&r.height>0;
  }
  function audit(){
    const sheet=document.getElementById?.('v5Sheet');
    const viewport={
      width:window.innerWidth||document.documentElement?.clientWidth||0,
      height:window.innerHeight||document.documentElement?.clientHeight||0,
      dpr:window.devicePixelRatio||1
    };
    const issues=[];
    const details={roots:0,arabic:0,buttons:0};

    if(sheet&&visible(sheet)){
      const sr=sheet.getBoundingClientRect();
      if(sr.left<-1||sr.right>viewport.width+1)issues.push({type:'sheet-horizontal-overflow',left:sr.left,right:sr.right,viewport:viewport.width});
      if(sr.height>viewport.height+1)issues.push({type:'sheet-vertical-overflow',height:sr.height,viewport:viewport.height});
      if(sheet.scrollWidth>sheet.clientWidth+2)issues.push({type:'sheet-scrollwidth',scrollWidth:sheet.scrollWidth,clientWidth:sheet.clientWidth});
    }

    for(const sel of ROOT_SELECTORS){
      for(const el of document.querySelectorAll?.(sel)||[]){
        if(!visible(el))continue;
        details.roots++;
        if(el.scrollWidth>el.clientWidth+2)issues.push({type:'content-horizontal-overflow',selector:sel,scrollWidth:el.scrollWidth,clientWidth:el.clientWidth});
      }
    }

    for(const sel of ARABIC_SELECTORS){
      for(const el of document.querySelectorAll?.(sel)||[]){
        if(!visible(el))continue;
        details.arabic++;
        const cs=window.getComputedStyle?.(el);
        if(cs&&cs.direction!=='rtl')issues.push({type:'arabic-not-rtl',selector:sel,direction:cs.direction});
        if(el.scrollWidth>el.clientWidth+2)issues.push({type:'arabic-overflow',selector:sel});
      }
    }

    for(const sel of BUTTON_SELECTORS){
      for(const el of document.querySelectorAll?.(sel)||[]){
        if(!visible(el))continue;
        details.buttons++;
        const r=el.getBoundingClientRect();
        if(r.height<40)issues.push({type:'small-tap-target',selector:sel,height:Math.round(r.height)});
        if(r.right>viewport.width+1||r.left<-1)issues.push({type:'button-horizontal-overflow',selector:sel});
      }
    }

    const result={
      ok:issues.length===0,
      viewport,
      issues,
      details,
      at:new Date().toISOString()
    };
    document.documentElement.dataset.deenQuranDeviceQa=result.ok?'pass':'warn';
    window.__deenLastQuranDeviceAudit=result;
    return result;
  }

  let timer=null;
  function schedule(){
    if(timer)clearTimeout(timer);
    timer=setTimeout(()=>{timer=null;audit()},120);
  }

  document.addEventListener?.('deen:navigated',schedule);
  document.addEventListener?.('click',e=>{
    if(e.target?.closest?.('#v5Sheet,.deen-quran-quick,#leagueScreen'))schedule();
  },true);
  window.addEventListener?.('resize',schedule);
  window.addEventListener?.('orientationchange',schedule);

  if(typeof MutationObserver!=='undefined'){
    const obs=new MutationObserver(m=>{if(m.some(x=>x.addedNodes?.length||x.removedNodes?.length))schedule()});
    const root=document.getElementById?.('v5Sheet')||document.body;
    if(root)obs.observe(root,{childList:true,subtree:true});
  }

  window.DEEN_QURAN_DEVICE_QA={audit,schedule,last:()=>window.__deenLastQuranDeviceAudit||null};
  schedule();
})();
</script>
