<!-- DEEN v10.5.3 — Isolated matching interaction engine -->
<script id="deen-v1053-isolated-matching-engine-js">
(()=>{
  if(window.__deenV1053IsolatedMatchingEngine)return;
  window.__deenV1053IsolatedMatchingEngine=true;

  const VERSION='10.5.3-ISOLATED-MATCHING';
  const BUILD='9153isolatedmatch1';

  let state=null;
  let clicks=0;
  let correctPairs=0;
  let wrongPairs=0;
  let arms=0;

  const norm=v=>String(v??'')
    .toLocaleLowerCase('tr-TR')
    .trim()
    .replace(/[.,;:!?'"“”‘’()]/g,'')
    .replace(/\s+/g,' ');

  function question(){
    try{return session?.questions?.[Number(session?.index)||0]||null}catch(_){return null}
  }

  function active(){
    const q=question();
    return !!q &&
      String(q.activity_type||'').toLowerCase()==='matching' &&
      Array.isArray(q.pairs) &&
      !!document.querySelector('#lesson .v1017-shell');
  }

  function key(){
    const q=question();
    let idx=0;
    try{idx=Number(session?.index)||0}catch(_){}
    let run='lesson';
    try{
      if(session && !session.__deenV1053RunId){
        session.__deenV1053RunId='m-'+Date.now()+'-'+Math.random().toString(36).slice(2,7);
      }
      run=session?.__deenV1053RunId||run;
    }catch(_){}
    return run+'::'+idx+'::'+String(q?.id||q?.question_id||q?.question_tr||'matching');
  }

  function sideOf(card){
    if(!card)return null;
    if(card.closest('.v1017-left'))return'left';
    if(card.closest('.v1017-right'))return'right';
    return null;
  }

  function textOf(card){
    return String(card?.querySelector?.('.v1017-card-label')?.textContent||card?.textContent||'').trim();
  }

  function descriptor(card){
    const side=sideOf(card);
    if(!side)return null;
    return {
      side,
      pair:String(card.dataset.pair??''),
      text:textOf(card),
      token:side+'::'+String(card.dataset.pair??'')+'::'+norm(textOf(card))
    };
  }

  function ensureState(){
    if(!active())return null;
    const k=key();
    if(!state||state.key!==k){
      state={
        key:k,
        selected:null,
        matchedLeft:new Set(),
        matchedRight:new Set(),
        errors:0,
        finished:false
      };
    }
    return state;
  }

  function cards(){
    return [...document.querySelectorAll('#lesson .v1017-card')];
  }

  function findCard(desc){
    if(!desc)return null;
    return cards().find(card=>{
      const d=descriptor(card);
      return d &&
        d.side===desc.side &&
        d.pair===desc.pair &&
        norm(d.text)===norm(desc.text);
    })||null;
  }

  function clearSelectedVisual(){
    cards().forEach(card=>card.classList.remove('selected'));
  }

  function paintSelected(card){
    clearSelectedVisual();
    if(card&&!card.disabled&&!card.classList.contains('matched')){
      card.classList.add('selected');
    }
  }

  function paintMatched(card){
    if(!card)return;
    card.classList.remove('selected','wrong');
    card.classList.add('matched');
    card.disabled=true;
    const indicator=card.querySelector('.v1017-choice-indicator');
    if(indicator)indicator.textContent='✓';
  }

  function setIndicatorNeutral(card){
    if(!card||card.classList.contains('matched'))return;
    const indicator=card.querySelector('.v1017-choice-indicator');
    if(indicator)indicator.textContent='';
  }

  function updateFooter(s=ensureState()){
    if(!s)return;
    const q=question();
    const total=Array.isArray(q?.pairs)?q.pairs.length:0;
    const done=s.matchedLeft.size;
    const count=document.querySelector('#lesson .v1017-count');
    const fill=document.querySelector('#lesson .v1017-meter i');
    if(count)count.textContent=done+' / '+total+' eşleşme';
    if(fill)fill.style.width=(total?Math.round(done/total*100):0)+'%';
    document.documentElement.dataset.deenMatchingProgress=done+'/'+total;
  }

  function rehydrate(){
    if(!active())return false;
    const s=ensureState();
    if(!s)return false;

    cards().forEach(card=>{
      const d=descriptor(card);
      if(!d)return;
      card.onclick=null;
      card.removeAttribute('onclick');
      card.dataset.deenMatchEngine='v1053';

      if(d.side==='left'&&s.matchedLeft.has(d.token))paintMatched(card);
      else if(d.side==='right'&&s.matchedRight.has(d.token))paintMatched(card);
      else{
        card.disabled=false;
        card.classList.remove('matched','wrong');
        setIndicatorNeutral(card);
      }
    });

    clearSelectedVisual();
    if(s.selected){
      const selected=findCard(s.selected);
      if(selected&&!selected.disabled)selected.classList.add('selected');
    }

    updateFooter(s);
    document.documentElement.dataset.deenMatchingInteraction='isolated-v1053';
    return true;
  }

  function expectedRight(leftDesc){
    const q=question();
    if(!q||!Array.isArray(q.pairs)||!leftDesc)return null;
    const pair=q.pairs.find(p=>norm(p?.left)===norm(leftDesc.text));
    return pair?String(pair.right??''):null;
  }

  function correct(leftDesc,rightDesc){
    const expected=expectedRight(leftDesc);
    return expected!==null && norm(expected)===norm(rightDesc?.text);
  }

  function select(card,s){
    const d=descriptor(card);
    if(!d)return;
    s.selected=d;
    paintSelected(card);
    document.documentElement.dataset.deenMatchingLast='selected';
  }

  function clearSelection(s){
    s.selected=null;
    clearSelectedVisual();
  }

  function finish(s){
    const q=question();
    if(!s||!q||s.finished)return;
    if(s.matchedLeft.size!==q.pairs.length)return;

    s.finished=true;
    clearSelection(s);
    updateFooter(s);
    document.documentElement.dataset.deenMatchingFinished='1';

    const savedKey=s.key;
    setTimeout(()=>{
      if(!state||state.key!==savedKey)return;
      try{window.completeQuestion?.(s.errors===0,q)}
      catch(err){console.error('DEEN isolated matching completion failed',err)}
    },220);
  }

  function good(leftDesc,rightDesc,s){
    const leftCard=findCard(leftDesc);
    const rightCard=findCard(rightDesc);
    if(!leftCard||!rightCard)return;

    try{navigator.vibrate?.(14)}catch(_){}

    s.matchedLeft.add(leftDesc.token);
    s.matchedRight.add(rightDesc.token);
    s.selected=null;
    correctPairs++;

    paintMatched(leftCard);
    paintMatched(rightCard);
    updateFooter(s);
    document.documentElement.dataset.deenMatchingLast='correct';

    finish(s);
  }

  function bad(leftDesc,rightDesc,s){
    const leftCard=findCard(leftDesc);
    const rightCard=findCard(rightDesc);

    s.errors++;
    s.selected=null;
    wrongPairs++;
    clearSelectedVisual();

    try{navigator.vibrate?.([10,22,10])}catch(_){}

    [leftCard,rightCard].forEach(card=>card?.classList?.add('wrong'));
    setTimeout(()=>{
      [leftCard,rightCard].forEach(card=>card?.classList?.remove('wrong'));
    },300);

    document.documentElement.dataset.deenMatchingLast='wrong';
  }

  function handle(card){
    if(!active()||!card||card.disabled)return false;

    const s=ensureState();
    const current=descriptor(card);
    if(!s||!current||s.finished)return false;

    clicks++;

    if(!s.selected){
      select(card,s);
      return true;
    }

    const previous=s.selected;

    if(previous.token===current.token){
      clearSelection(s);
      document.documentElement.dataset.deenMatchingLast='cleared';
      return true;
    }

    if(previous.side===current.side){
      select(card,s);
      document.documentElement.dataset.deenMatchingLast='reselected';
      return true;
    }

    const left=previous.side==='left'?previous:current;
    const right=previous.side==='right'?previous:current;

    if(correct(left,right))good(left,right,s);
    else bad(left,right,s);

    return true;
  }

  function arm(){
    if(!active())return false;
    arms++;
    ensureState();
    cards().forEach(card=>{
      card.onclick=null;
      card.removeAttribute('onclick');
      card.dataset.deenMatchEngine='v1053';
    });
    rehydrate();
    return true;
  }

  document.addEventListener('click',event=>{
    const card=event.target?.closest?.('#lesson .v1017-card');
    if(!card||!active())return;

    event.preventDefault();
    event.stopPropagation();
    event.stopImmediatePropagation();

    handle(card);
  },true);

  document.addEventListener('deen:question-rendering',()=>{
    const q=question();
    if(!q||String(q.activity_type||'').toLowerCase()!=='matching'){
      state=null;
      document.documentElement.dataset.deenMatchingInteraction='idle';
    }
  });

  document.addEventListener('deen:question-rendered',event=>{
    const run=()=>arm();
    if(!window.DEEN_RUNTIME_SCHEDULE?.question?.(
      'v1053-isolated-matching',
      event?.detail,
      run,
      {raf:true,delays:[0,80,220]}
    )){
      requestAnimationFrame(run);
      setTimeout(run,80);
      setTimeout(run,220);
    }
  });

  window.DEEN_ISOLATED_MATCHING={
    version:VERSION,
    build:BUILD,
    arm,
    rehydrate,
    snapshot:()=>({
      version:VERSION,
      build:BUILD,
      active:active(),
      key:state?.key||null,
      selected:state?.selected?{...state.selected}:null,
      matchedLeft:state?.matchedLeft?.size||0,
      matchedRight:state?.matchedRight?.size||0,
      errors:state?.errors||0,
      finished:state?.finished||false,
      clicks,
      correctPairs,
      wrongPairs,
      arms,
      renderer:document.documentElement.dataset.deenMatchingRenderer||null
    })
  };
})();
</script>
