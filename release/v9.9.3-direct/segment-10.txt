<script id="deen-v530-adaptive-lesson-composer">
/* DEEN v5.3.0
   Adaptive Lesson Composer.
   Visual theme, question data, sound/game-feel and final selectors are preserved. */
(() => {
  const V530 = {
    version: '5.3.0',
    maxRecentPerStage: 10,
    profileFor(stageId){
      const st=STAGES.find(s=>s.id===stageId);
      const unit=stageId.slice(0,3);
      const score=Number(state.stageScores?.[stageId]);
      const mastery=Number(masteryFor(unit)||0);
      const challenge=stageId.endsWith('S07');
      const support=(Number.isFinite(score)&&score<72)||mastery<42;
      const stretch=(Number.isFinite(score)&&score>=92&&mastery>=78)||challenge;
      return {challenge,support,stretch,score:Number.isFinite(score)?score:null,mastery,stage:st};
    },
    slotProfiles(target, profile){
      if(target===10){
        return [
          {cog:['recognition'],diff:['easy','medium'],role:'warmup'},
          {cog:['recall','recognition'],diff:['easy','medium'],role:'recall'},
          {types:['matching','fill_blank','missing_item','true_false','speed_review','build_sequence'],role:'interaction'},
          {cog:['application'],types:['scenario','error_spotting','multiple_choice'],role:'apply'},
          {cog:['recall','application'],types:['progressive_recall','fill_blank','build_sequence','matching','missing_item'],role:'retrieve'},
          {cog:['application','recall'],diff:['medium','hard'],role:'mixed'},
          {priority:true,role:'weakness'},
          {diff:['medium','hard'],cog:['recall','application'],role:'stretch'},
          {diff:['hard','medium'],cog:['application','recall'],role:'challenge'},
          {diff:['hard','medium'],cog:['application','recall'],priority:true,role:'finish'}
        ];
      }
      return [
        {cog:['recognition'],diff:profile.support?['easy']:['easy','medium'],role:'warmup'},
        {cog:['recognition','recall'],diff:['easy','medium'],role:'recognize'},
        {types:['matching','fill_blank','missing_item','true_false','speed_review','build_sequence'],role:'interaction'},
        {cog:['recall','application'],types:['progressive_recall','fill_blank','matching','build_sequence','scenario','multiple_choice'],role:'retrieve'},
        {cog:['application','recall'],types:['scenario','error_spotting','multiple_choice','missing_item'],role:'apply'},
        {priority:true,cog:['recall','application','recognition'],role:'adaptive'},
        {cog:['application','recall'],diff:profile.stretch?['hard','medium']:['medium','hard','easy'],role:'finish'}
      ];
    },
    recent(stageId){
      const h=state.lessonHistory?.[stageId];
      return Array.isArray(h)?h:[];
    },
    ensureState(){
      if(!state.lessonHistory||typeof state.lessonHistory!=='object'||Array.isArray(state.lessonHistory)) state.lessonHistory={};
    },
    remember(stageId, ids){
      this.ensureState();
      const prev=this.recent(stageId);
      const next=[...ids,...prev.filter(id=>!ids.includes(id))].slice(0,this.maxRecentPerStage);
      state.lessonHistory[stageId]=next;
    },
    priorityBoost(q){
      const r=state.reviewSchedule?.[q.id];
      let b=1;
      if(state.mistakes?.includes(q.id)) b*=3.4;
      if(r?.nextReview<=Date.now()) b*=2.1;
      const ss=Number(state.stageScores?.[q.stage_id]);
      if(Number.isFinite(ss)&&ss<75) b*=1.25;
      return b;
    },
    fit(q,slot,ctx){
      let f=1;
      const profile=ctx.profile;
      if(slot.cog?.length){
        const idx=slot.cog.indexOf(q.cognitive_level);
        f*=idx<0?.52:(1.85-idx*.25);
      }
      if(slot.diff?.length){
        const idx=slot.diff.indexOf(q.difficulty);
        f*=idx<0?.60:(1.65-idx*.18);
      }
      if(slot.types?.length){
        const idx=slot.types.indexOf(q.activity_type);
        f*=idx<0?.58:(1.7-idx*.08);
      }
      if(slot.priority) f*=this.priorityBoost(q);

      // Adapt difficulty without making struggling users feel punished.
      if(profile.support){
        if(q.difficulty==='easy')f*=1.35;
        if(q.difficulty==='hard')f*=.55;
      }else if(profile.stretch){
        if(q.difficulty==='hard')f*=1.35;
        else if(q.difficulty==='medium')f*=1.15;
      }

      const tc=ctx.typeCounts[q.activity_type]||0;
      const cc=ctx.conceptCounts[q.concept_id]||0;
      const vg=q.variant_group_id||'';
      const vc=vg?(ctx.variantCounts[vg]||0):0;

      // Avoid monotonous lessons.
      f/=1+tc*.48;
      f/=1+Math.max(0,cc-1)*.70;
      if(vc)f*=.20;

      const last=ctx.out[ctx.out.length-1];
      const prev=ctx.out[ctx.out.length-2];
      if(last?.activity_type===q.activity_type)f*=.40;
      if(last?.activity_type===q.activity_type&&prev?.activity_type===q.activity_type)f*=.08;
      if(last?.concept_id===q.concept_id)f*=.58;

      // Replays should feel fresh, but weak/mistake questions can still return.
      if(ctx.recent.has(q.id)){
        f*=state.mistakes?.includes(q.id)?.72:.18;
      }

      return Math.max(.005, f);
    },
    pick(pool,slot,ctx){
      let candidates=pool.filter(q=>!ctx.used.has(q.id));
      if(!candidates.length)return null;

      // Hard constraints are applied only when the pool can satisfy them.
      if(slot.forceHard){
        const hard=candidates.filter(q=>q.difficulty==='hard');
        if(hard.length)candidates=hard;
      }
      if(slot.newType){
        const freshType=candidates.filter(q=>!(ctx.typeCounts[q.activity_type]||0));
        if(freshType.length)candidates=freshType;
      }

      // Never allow three identical interaction types consecutively when an alternative exists.
      const last=ctx.out[ctx.out.length-1],prev=ctx.out[ctx.out.length-2];
      if(last&&prev&&last.activity_type===prev.activity_type){
        const alternate=candidates.filter(q=>q.activity_type!==last.activity_type);
        if(alternate.length)candidates=alternate;
      }

      // Normal lesson replays strongly prefer unseen questions.
      // Smart Review is allowed to intentionally repeat due/mistake items.
      if(!ctx.reviewMode&&ctx.recent?.size){
        const unseen=candidates.filter(q=>!ctx.recent.has(q.id));
        if(unseen.length)candidates=unseen;
      }

      return weightedChoice(candidates,q=>Math.max(.01,weight(q))*this.fit(q,slot,ctx));
    },
    compose(stageId){
      const target=stageId.endsWith('S07')?10:7;
      const pool=poolFor(stageId).filter(q=>!audioBlocked(q));
      if(pool.length<target)return [];
      const profile=this.profileFor(stageId);
      const slots=this.slotProfiles(target,profile);
      const ctx={
        out:[],used:new Set(),recent:new Set(this.recent(stageId)),
        typeCounts:{},conceptCounts:{},variantCounts:{},profile,reviewMode:false
      };
      const desiredDiversity=Math.min(target===10?5:3,new Set(pool.map(q=>q.activity_type)).size);
      const hardAvailable=profile.challenge&&pool.some(q=>q.difficulty==='hard');
      for(let si=0;si<slots.length;si++){
        const slot={...slots[si]};
        const distinct=Object.keys(ctx.typeCounts).length;
        const remaining=slots.length-si;
        const diversityNeeded=desiredDiversity-distinct;
        if(diversityNeeded>=remaining)slot.newType=true;
        if(hardAvailable&&si===slots.length-1&&!ctx.out.some(q=>q.difficulty==='hard'))slot.forceHard=true;

        let q=this.pick(pool,slot,ctx);
        if(!q)q=this.pick(pool,{newType:slot.newType,forceHard:slot.forceHard},ctx);
        if(!q)break;
        ctx.out.push(q);ctx.used.add(q.id);
        ctx.typeCounts[q.activity_type]=(ctx.typeCounts[q.activity_type]||0)+1;
        ctx.conceptCounts[q.concept_id]=(ctx.conceptCounts[q.concept_id]||0)+1;
        if(q.variant_group_id)ctx.variantCounts[q.variant_group_id]=(ctx.variantCounts[q.variant_group_id]||0)+1;
      }
      while(ctx.out.length<target){
        const q=this.pick(pool,{priority:true},ctx);
        if(!q)break;
        ctx.out.push(q);ctx.used.add(q.id);
        ctx.typeCounts[q.activity_type]=(ctx.typeCounts[q.activity_type]||0)+1;
        ctx.conceptCounts[q.concept_id]=(ctx.conceptCounts[q.concept_id]||0)+1;
        if(q.variant_group_id)ctx.variantCounts[q.variant_group_id]=(ctx.variantCounts[q.variant_group_id]||0)+1;
      }
      return this.repairSequence(ctx.out,pool,target,profile);
    },
    repairSequence(out,pool,target,profile){
      const arr=[...out];
      const ids=new Set(arr.map(q=>q.id));
      const typeCounts=()=>arr.reduce((m,q)=>(m[q.activity_type]=(m[q.activity_type]||0)+1,m),{});

      // Guarantee minimum activity diversity when the stage pool supports it.
      let guard=0;

        const present=new Set(arr.map(q=>q.activity_type));
        const candidates=pool.filter(q=>!ids.has(q.id)&&!present.has(q.activity_type));
        if(!candidates.length)break;
        const q=weightedChoice(candidates,x=>Math.max(.01,weight(x)));
        const tc=typeCounts();
        let idx=-1;
        for(let i=arr.length-1;i>=1;i--)if(tc[arr[i].activity_type]>1){idx=i;break}
        if(idx<0)break;
        ids.delete(arr[idx].id);arr[idx]=q;ids.add(q.id);
      }

      // Challenge lessons must contain a hard item when the source pool has one.

        const last=reordered[reordered.length-1]?.activity_type;
        const prev=reordered[reordered.length-2]?.activity_type;
        const blocked=(last&&prev&&last===prev)?last:null;
        let types=Object.keys(grouped).filter(t=>grouped[t].length&&t!==blocked);
        if(!types.length)types=Object.keys(grouped).filter(t=>grouped[t].length);
        types.sort((a,b)=>grouped[b].length-grouped[a].length);
        const chosen=types[0];
        reordered.push(grouped[chosen].shift());
      }
      return reordered;
    },

        {types:['matching','fill_blank','missing_item','progressive_recall','build_sequence'],role:'retrieve'},
        {cog:['application'],types:['scenario','error_spotting','multiple_choice'],role:'apply'},

        ctx.typeCounts[q.activity_type]=(ctx.typeCounts[q.activity_type]||0)+1;
        ctx.conceptCounts[q.concept_id]=(ctx.conceptCounts[q.concept_id]||0)+1;
        if(q.variant_group_id)ctx.variantCounts[q.variant_group_id]=(ctx.variantCounts[q.variant_group_id]||0)+1;
      }
      while(ctx.out.length<target){
        const q=this.pick(clean,{priority:true},ctx);
        if(!q)break;
        ctx.out.push(q);ctx.used.add(q.id);
        ctx.typeCounts[q.activity_type]=(ctx.typeCounts[q.activity_type]||0)+1;
        ctx.conceptCounts[q.concept_id]=(ctx.conceptCounts[q.concept_id]||0)+1;
      }
      return this.repairSequence(ctx.out,clean,target,{challenge:false,support:false,stretch:false});
    }
  };

    if(final2Unlocked()&&!state.section2Final.completed)return {type:'final2',title:'İman Ustalığı hazır',detail:'Kısım 2 bilgilerini bir araya getir.',action:"startSection2Final()",cta:'FİNALE GİR'};
    return null;
  }
  function smartReturnDecision(){
    const {d,r,w}=dailyStatus(),next=currentStage(),due=dueCount();
    const inactive=daysSince(state.streakMeta?.lastStudyDate);
    const firstToday=(d.lessons||0)===0;
    const goalDone=(d.lessons||0)>=state.dailyGoal;
    const allDone=goalDone&&r.complete;
    const final=sectionRecentlyUnlocked();

    if(final&&firstToday)return {...final,reason:'milestone'};

    if(firstToday&&!state.streakMeta?.lastStudyDate&&!(state.history?.length||0)&&!(state.completed?.length||0)&&next){
      return {
        type:'first_start',title:'Yolculuğuna başla.',
        detail:`${next.title} · 7 soru · yaklaşık 3 dakika.`,
        action:`scrollToCurrent('${next.unit}');preview('${next.id}')`,
        cta:'İLK DERSE BAŞLA'
      };
    }

    const newUnit=newlyOpenedUnit(next);
    if(newUnit&&firstToday)return newUnit;

    // After a few inactive days, reduce friction instead of talking about a lost streak.
    if(firstToday&&inactive>=3){
      if(due>=5)return {
        type:'return_review',title:'Tekrar başlayalım.',
        detail:`${due} soru tekrar bekliyor. 5 soruluk kısa bir ısınmayla geri dön.`,
        action:"startSmartReview('quick')",cta:'5 SORULUK ISINMA',soft:'Kaldığın yol burada. Seri baskısı yok.'
      };
      if(next)return {
        type:'return_lesson',title:'Tekrar başlayalım.',
        detail:`${next.title} · yaklaşık 5 dakika.`,
        action:`scrollToCurrent('${next.unit}');preview('${next.id}')`,cta:'KISA DERSE BAŞLA',soft:'Kaldığın yerden devam et.'
      };
    }

    // First session: lots of due material gets a quick warm-up before new content.
    if(firstToday&&due>=8){
      return {
        type:'warmup',title:'Önce kısa bir ısınma.',
        detail:`${due} tekrar bekliyor. 5 soruyla hafızanı tazele, sonra yeni derse geç.`,
        action:"startSmartReview('quick')",cta:'5 SORULUK ISINMA',soft:'Yaklaşık 2 dakika.'
      };
    }

    // Maintain an active streak with one obvious next lesson.
    if(firstToday&&state.streak>0&&next){
      const st=currentState(next);
      const challenge=next.id.endsWith('S07')||next.kind==='challenge';
      return {
        type:challenge?'challenge':'streak',
        title:challenge?'Challenge hazır.':`${state.streak} günlük serini bugün de sürdür.`,
        detail:`${next.title} · ${challenge?'10 soru':'7 soru'} · yaklaşık ${challenge?'4':'3'} dakika.`,
        action:`scrollToCurrent('${next.unit}');preview('${next.id}')`,
        cta:challenge?'CHALLENGE’A GİR':'SERİYİ SÜRDÜR'
      };
    }

    // After the daily lesson target, due review becomes the best next action.
    if(goalDone&&due>0&&!allDone){
      return {
        type:'review',title:'Günlük ders hedefin tamam.',
        detail:`Şimdi ${due} bekleyen sorudan kısa bir tekrar yaparak bilgiyi sabitle.`,
        action:"startSmartReview('due')",cta:'KISA TEKRAR'
      };
    }

    if(allDone){
      return {
        type:'done',title:'Bugünün yolculuğu tamamlandı.',
        detail:'Ders hedefin ve günlük görevlerin bitti. İstersen kısa bir tekrar yapabilirsin.',
        action:"startSmartReview('quick')",cta:'İSTEĞE BAĞLI TEKRAR',soft:'Bugün daha fazla oynaman gerekmiyor.'
      };
    }

    if(final)return {...final,reason:'milestone'};
    if(next){
      const challenge=next.id.endsWith('S07')||next.kind==='challenge';
      return {
        type:challenge?'challenge':'lesson',
        title:challenge?'Challenge hazır.':'Kaldığın yerden devam et.',
        detail:`${next.title} · ${challenge?'10 soru':'7 soru'} · yaklaşık ${challenge?'4':'3'} dakika.`,
        action:`scrollToCurrent('${next.unit}');preview('${next.id}')`,
        cta:challenge?'CHALLENGE’A GİR':'DEVAM ET'
      };
    }
    return {
      type:'review',title:'Bugün bilgini tazele.',
      detail:'Kısa bir tekrar oturumuyla öğrendiklerini canlı tut.',
      action:"startSmartReview('quick')",cta:'HIZLI TEKRAR'
    };
  }

  function journeyMarkup(){
    const {d,r,w}=dailyStatus(),due=dueCount();
    return `<small class="v55-return-summary">Görevler ${r.done}/3 · Haftalık ritim ${Math.min(w.days,w.goal)}/${w.goal} gün${due?' · '+due+' tekrar':''}</small>`;
  }

  returnCard=function(){
    const decision=smartReturnDecision(),{d}=dailyStatus();
    $('v5Return').innerHTML=`<section class="lp-return ${decision.type==='done'?'v55-done':''}">
      <div class="v5-row"><span class="v5-kicker">${decision.type==='done'?'BUGÜN TAMAM':'SIRADAKİ ADIM'}</span><b>${Math.min(d.lessons,state.dailyGoal)} / ${state.dailyGoal}</b></div>
      <h1>${esc(decision.title)}</h1>
      <p>${esc(decision.detail)}</p>
      ${primary(decision.cta,decision.action)}
      ${journeyMarkup()}
    </section>`;
  };

  function recommendationAfterLesson(){
    const {d,r}=dailyStatus(),due=dueCount(),next=currentStage();
    const goalDone=d.lessons>=state.dailyGoal;
    const allDone=goalDone&&r.complete;

    if(allDone)return {
      kind:'done',title:'Bugünün yolculuğu tamamlandı.',
      detail:'Ders hedefin ve 3 günlük görevin bitti.',
      primary:{label:'ÖĞRENME YOLUNA DÖN',action:"closeLesson();navigate('learnScreen')"},
      secondary:due?{label:'İSTEĞE BAĞLI TEKRAR',action:"closeLesson();startSmartReview('quick')"}:null
    };
    if(goalDone&&due>0)return {
      kind:'review',title:'Şimdi hatırlamayı güçlendir.',
      detail:`${due} soru tekrar bekliyor.`,
      primary:{label:'KISA TEKRAR',action:"closeLesson();startSmartReview('due')"},
      secondary:next?{label:'SONRAKİ DERS',action:`closeLesson();navigate('learnScreen');setTimeout(()=>{scrollToCurrent('${next.unit}');preview('${next.id}')},80)`}:null

        title:newUnit?newUnit.title:challenge?'Challenge hazır.':'Bir adım daha?',
        detail:newUnit?newUnit.detail:`${next.title} · ${challenge?'10':'7'} soru.`,
