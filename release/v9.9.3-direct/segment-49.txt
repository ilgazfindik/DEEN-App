 if(stage==='final'){if(!m.finalUnlocked)return false;return startQuiz('final')}
 if(stage==='mastery'){if(!m.state.final.passed)return false;return DEEN_QURAN_SRS?.openReview?.()||DEEN_QURAN_MEM_V2?.openWeak?.()}
 return false;
}
function allWords(id){const out=[];for(let n=1;n<=COUNTS[id];n++)out.push(...words(rec(id,n)?.arabic?.text));return[...new Set(out)]}
function optionSet(correct,candidates,seed){
 const uniq=[correct,...candidates.filter(x=>x&&x!==correct)].filter((x,i,a)=>a.indexOf(x)===i).slice(0,4);
 return seededShuffle(uniq.map((text,i)=>({id:'o'+i,text})),seed);
}
function qFill(id,n,seed){
 const r=rec(id,n),w=words(r?.arabic?.text);if(w.length<2)return qNumber(id,n,seed);
 const bi=hash(seed)%w.length,answer=w[bi],pool=[...allWords(id),...w].filter(x=>x!==answer);
 const display=w.map((x,i)=>i===bi?'_____':x).join(' ');
 return{kind:'fill',ayah:n,title:'Eksik kelimeyi tamamla',display,type:'arabic',answer,options:optionSet(answer,pool,seed)};
}
function qNext(id,n,seed){
 const total=COUNTS[id],base=Math.min(Math.max(1,n),Math.max(1,total-1)),target=base+1;
 const a=rec(id,base),answer=rec(id,target)?.arabic?.text;
 const pool=Array.from({length:total},(_,i)=>rec(id,i+1)?.arabic?.text).filter(x=>x!==answer);
 return{kind:'next',ayah:base,title:'Bu ayetten sonra hangisi gelir?',display:a?.arabic?.text,type:'arabic',answer,options:optionSet(answer,pool,seed)};
}
function qListen(id,n,seed){
 // Only verified local ayah audio is eligible. Never fabricate an audio source.
 const engine=window.DEEN_QURAN_AYAH_AUDIO,config=engine?.config?.(id),r=rec(id,n),answer=r?.arabic?.text;
 if(!config?.files?.[n-1]||!answer||typeof engine?.play!=='function')return qNext(id,n,seed);
 const pool=Array.from({length:COUNTS[id]},(_,i)=>rec(id,i+1)?.arabic?.text).filter(x=>x&&x!==answer);
 return{kind:'listen',ayah:n,title:'Dinlediğin ayeti seç',display:'',type:'audio',answer,options:optionSet(answer,pool,seed),optionsArabic:true};
}
function qContinuation(id,n,seed){
 const r=rec(id,n),w=words(r?.arabic?.text);if(w.length<3)return qFill(id,n,seed);
 const cut=Math.max(1,Math.floor(w.length/2)),display=w.slice(0,cut).join(' ')+' …',answer=w.slice(cut).join(' ');
 const pool=[];for(let x=1;x<=COUNTS[id];x++){if(x===n)continue;const z=words(rec(id,x)?.arabic?.text);pool.push(z.slice(Math.max(1,Math.floor(z.length/2))).join(' '))}
 return{kind:'continuation',ayah:n,title:'Ayetin doğru devamını seç',display,type:'arabic',answer,options:optionSet(answer,pool,seed)};
}
function qMealToAyah(id,n,seed){
 const r=rec(id,n),answer=r?.arabic?.text,pool=Array.from({length:COUNTS[id]},(_,i)=>rec(id,i+1)?.arabic?.text).filter(x=>x!==answer);
 return{kind:'mealToAyah',ayah:n,title:'Bu meal hangi ayete ait?',display:r?.translation?.text,type:'meal',answer,options:optionSet(answer,pool,seed),optionsArabic:true};
}
function qAyahToMeal(id,n,seed){
 const r=rec(id,n),answer=r?.translation?.text,pool=Array.from({length:COUNTS[id]},(_,i)=>rec(id,i+1)?.translation?.text).filter(x=>x!==answer);
 return{kind:'ayahToMeal',ayah:n,title:'Bu ayetin doğrulanmış meali hangisi?',display:r?.arabic?.text,type:'arabic',answer,options:optionSet(answer,pool,seed)};
}
function qNumber(id,n,seed){
 const r=rec(id,n),answer='Ayet '+n,pool=Array.from({length:COUNTS[id]},(_,i)=>'Ayet '+(i+1)).filter(x=>x!==answer);
 return{kind:'number',ayah:n,title:'Bu sure içinde kaçıncı ayet?',display:r?.arabic?.text,type:'arabic',answer,options:optionSet(answer,pool,seed)};
}
function weakOrder(id){
 const rows=DEEN_QURAN_MEM_V2?.weak?.(99)?.filter(x=>x.id===id)||[];
 const nums=rows.map(x=>x.n);for(let n=1;n<=COUNTS[id];n++)if(!nums.includes(n))nums.push(n);return nums;
}
function buildQuestions(stage,attempt){
 const id=currentSurah,total=COUNTS[id],seed=id+':'+stage+':'+attempt,weak=weakOrder(id),out=[];
 const pick=i=>weak[i%weak.length]||((i%total)+1);
 if(stage==='ayah'){
  const f=[qFill,qListen,qContinuation,qFill,qNext];
  for(let i=0;i<5;i++)out.push(f[i](id,pick(i),seed+':'+i));
 }
 if(stage==='meaning'){
  const f=[qMealToAyah,qAyahToMeal,qNumber,qMealToAyah,qAyahToMeal];
  for(let i=0;i<5;i++)out.push(f[i](id,pick(i),seed+':'+i));
 }
 if(stage==='final'){
  const f=[qFill,qMealToAyah,qNext,qAyahToMeal,qContinuation,qNumber,qFill,qListen];
  for(let i=0;i<8;i++)out.push(f[i](id,pick(i),seed+':'+i));
 }
 return out;
}

async function playQuizAudio(){

 if(statusNode)statusNode.textContent='Ayet sesi hazırlanıyor…';

  engine.stop?.(); // Reuse and rewind the global audio player.
  const ok=await engine.play(q.ayah,currentSurah);

function dragMove(ev){

  node.className='deen-qj2-drag-ghost';node.setAttribute('aria-hidden','true');
  node.textContent=drag.text;
  const host=document.getElementById?.('v5Sheet')||document.body;
  host?.appendChild?.(node);

 }

 const slot=document.querySelector?.('.deen-qj2-quiz [data-qj2-fill-slot]');

function dragEnd(ev){

 const inSlot=moved&&!!dropSlotAt(ev.clientX,ev.clientY);

  if(ev.cancelable)ev.preventDefault();

  ignoreClick={id,until:Date.now()+650,index:quiz?.index,stage:quiz?.stage,surah:currentSurah};
 }

  quiz.selected=id;
  replaceQuiz();
  return true;
 }
 return false;
}

function finishQuiz(){

 // Review practice must not alter stage attempts, personal best, unlocks, or SRS.
 if(quiz.practice){
  quiz.result={score,passed,threshold,stage:quiz.stage,practice:true,newBest:false,best:null};
  return quiz.result;
 }
 const s=sstate(currentSurah),slot=s[quiz.stage],previousBest=slot.best,previousAttempts=slot.attempts;
 slot.attempts++;slot.best=Math.max(slot.best,score);slot.lastScore=score;slot.lastAt=new Date().toISOString();

 DEEN_QURAN.persistState?.();
 quiz.result={score,passed,threshold,stage:quiz.stage,practice:false,newBest:score>previousBest&&score>0,best:slot.best,previousAttempts};
 return quiz.result;
}
