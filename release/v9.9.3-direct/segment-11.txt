    const total = lessonTotalFor(stageId);
    const base = state.stageLessonMeta[stageId] || {};
    const meta = Object.assign({completedLessons:0,lessonScores:{},seenIds:[],lastLessonNo:1,totalLessons:total,lastPct:0}, base);
    meta.totalLessons = total;
    meta.completedLessons = Math.max(0, Math.min(total, Number(meta.completedLessons)||0));
    meta.lessonScores = meta.lessonScores && typeof meta.lessonScores==='object' ? meta.lessonScores : {};
    meta.seenIds = Array.isArray(meta.seenIds) ? meta.seenIds : [];
    meta.lastLessonNo = Math.max(1, Math.min(total, Number(meta.lastLessonNo)||1));
    meta.lastPct = Number(meta.lastPct)||0;
    state.stageLessonMeta[stageId] = meta;
    return meta;
  }
  function updateAggregateScore(stageId){
    const meta = getStageMeta(stageId);
    const vals = Object.values(meta.lessonScores||{}).map(Number).filter(Number.isFinite);
    if(vals.length){ state.stageScores[stageId] = Math.round(vals.reduce((a,b)=>a+b,0)/vals.length); }
  }
  function isStageComplete(stageId){
    const meta = getStageMeta(stageId);
    return meta.completedLessons >= meta.totalLessons && state.completed.includes(stageId);
  }
  function nextLessonNo(stageId){
    const meta = getStageMeta(stageId);
    if(isStageComplete(stageId)) return meta.totalLessons;
    return Math.min(meta.completedLessons + 1, meta.totalLessons);
  }
  function pickLessonRecipe(stageId, lessonNo){
    if(lessonTotalFor(stageId)===1) return {types:[], cognition:['recognition','recall','application'], difficulty:['easy','medium','hard']};
    return LESSON_RECIPES[Math.max(0, Math.min(LESSON_RECIPES.length-1, lessonNo-1))];
  }
  function questionScoreForRecipe(q, recipe, seenSet){
    let s = Math.random()*0.4;
    if(recipe.types.includes(q.activity_type)) s += 4.2;
    if(recipe.cognition.includes(q.cognitive_level)) s += 1.8;
    if(recipe.difficulty.includes(q.difficulty)) s += 1.2;
    if(state.mistakes && state.mistakes.includes(q.id)) s += 1.6;
    if(!seenSet.has(q.id)) s += 2.0; else s -= 1.5;
    if(q.activity_type==='listening' && (window.audioBlocked && audioBlocked(q))) s -= 99;
    return s;
  }
  function chooseMultiLessonQuestions(stageId, lessonNo){
    const totalLessons = lessonTotalFor(stageId);
    const target = totalLessons===1 ? lessonTargetFor(stageId) : (lessonNo>=4 ? 6 : 5);
    const meta = getStageMeta(stageId);
    const seenSet = new Set(meta.seenIds||[]);
    let pool = (typeof poolFor==='function' ? poolFor(stageId) : []).filter(q => !(window.audioBlocked && audioBlocked(q)));
    if(!pool.length) return [];
    const recipe = pickLessonRecipe(stageId, lessonNo);
    const groups = {};
    const uniqueConcepts = new Set();
    pool.sort((a,b)=>questionScoreForRecipe(b, recipe, seenSet) - questionScoreForRecipe(a, recipe, seenSet));
    const preferred = pool.filter(q => questionScoreForRecipe(q, recipe, seenSet) > 0.8);
    const basePool = preferred.length >= Math.min(target, pool.length) ? preferred : pool;
    basePool.forEach(q => { (groups[q.activity_type] = groups[q.activity_type] || []).push(q); });
    const picked = [];
    const usedIds = new Set();
    Object.keys(groups).sort((a,b)=>{
      const ap = recipe.types.indexOf(a), bp = recipe.types.indexOf(b);
      return (ap<0?99:ap) - (bp<0?99:bp);
    }).forEach(type => {
      const list = groups[type];
      if(!list || !list.length || picked.length>=target) return;
      const candidate = list[0];
      if(candidate && !usedIds.has(candidate.id)){
        picked.push(candidate); usedIds.add(candidate.id); if(candidate.concept_id) uniqueConcepts.add(candidate.concept_id);
      }
    });
    for(const q of basePool){
      if(picked.length>=target) break;
      if(usedIds.has(q.id)) continue;
      if(q.concept_id && uniqueConcepts.has(q.concept_id) && Math.random() < 0.35) continue;
      picked.push(q); usedIds.add(q.id); if(q.concept_id) uniqueConcepts.add(q.concept_id);
    }
    for(const q of pool){
      if(picked.length>=target) break;
      if(usedIds.has(q.id)) continue;
      picked.push(q); usedIds.add(q.id);
    }
    return (typeof shuffle==='function' ? shuffle(picked) : picked).slice(0,target);
  }

  const __oldStartStage = window.startStage;
  const __oldShowResult = window.showResult;

  function beginMultiLessonStage(stageId, lessonNoOverride, replayMode){
    ensureStateShape();
    const st = STAGES.find(x=>x.id===stageId);
    if(!st) return;
    const totalLessons = lessonTotalFor(stageId);
    const meta = getStageMeta(stageId);
    const lessonNo = Math.max(1, Math.min(totalLessons, Number(lessonNoOverride)||nextLessonNo(stageId)));
    const lessonEnergy = totalLessons===1 ? st.energy : Math.max(1, Math.min(3, Math.round(st.energy/2)));
    if(!state.premium && state.energy < lessonEnergy){ if(typeof showEnergy==='function') showEnergy(); return; }
    if(!state.premium) state.energy -= lessonEnergy;
    const qs = chooseMultiLessonQuestions(stageId, lessonNo);
    if(!Array.isArray(qs) || !qs.length){ console.error('Multi-lesson stage question pool empty', stageId, lessonNo); return; }
    meta.lastLessonNo = lessonNo;
    state.stageAttemptCounts[stageId] = (state.stageAttemptCounts[stageId]||0) + 1;
    qs.forEach(q=>{ if(!meta.seenIds.includes(q.id)) meta.seenIds.push(q.id); });
    session = {stage:deepClone(st),questions:qs,index:0,correct:0,combo:0,bestCombo:0,multiLessonStage:true,lessonNo,lessonTotal:totalLessons,lessonEnergy,replayMode:!!replayMode,stageMetaRef:meta};
    const lesson = document.getElementById('lesson');
