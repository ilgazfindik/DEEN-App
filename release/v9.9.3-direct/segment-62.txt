<script id="deen-v10974-quran-ayet-test-js">
(()=>{
if(window.__deenV10974QuranAyetTest)return;
window.__deenV10974QuranAyetTest=true;
const BUILD='9294ayettestmvp1';
const SURAHS={fatiha:{name:'Fâtiha',count:7},ikhlas:{name:'İhlâs',count:4}};
const TOTAL=5;
const STAGES=[
 {type:'fill',ayah:1},
 {type:'meaning',ayah:2},
 {type:'continuation',ayah:3},
 {type:'next',ayah:2},
 {type:'fill',ayah:0}
];
let run=null;
const esc=v=>String(v??'').replace(/[&<>"']/g,x=>({'&':'&amp;','<':'&lt;','>':'&gt;','"':'&quot;','\'':'&#39;'}[x]));
const words=s=>String(s||'').trim().split(/\s+/).filter(Boolean);
function shuffle(xs){
 const a=xs.slice();
 for(let i=a.length-1;i>0;i--){const j=Math.floor(Math.random()*(i+1));[a[i],a[j]]=[a[j],a[i]]}
 return a;
}
function registeredRows(id){
 if(!SURAHS[id])return [];
 const rows=[];
 for(let n=1;n<=SURAHS[id].count;n++){
  let rec=null;
  try{rec=window.DEEN_QURAN_CONTENT?.getAyah?.(id,n)}catch(_){}
  const ar=String(rec?.arabic?.text||'').trim();
  const tr=String(rec?.translation?.text||'').trim();
  if(!ar||!tr)continue;
  rows.push({id,ayah:n,arabic:ar,translation:tr});
 }
 return rows;
}
function poolFor(id){return registeredRows(id).concat(registeredRows(id==='fatiha'?'ikhlas':'fatiha'))}
function question(type,target,pool,all){
 let prompt='',answer='',opts=[],label='',arabicOpts=true;
 const ws=words(target.arabic);
 if(type==='fill'&&ws.length>=2){
  const n=ws.length===2?1:Math.floor(ws.length/2);
  answer=ws[n];prompt=ws.map((w,i)=>i===n?'__BLANK__':w);
  label='Eksik kelimeyi tamamla';
  const distractors=shuffle([...new Set(pool.flatMap(r=>words(r.arabic)).filter(w=>w!==answer))]).slice(0,3);
  opts=[answer,...distractors];
 }else if(type==='continuation'&&ws.length>=2){
  const n=Math.max(1,Math.floor(ws.length/2));
  prompt=ws.slice(0,n).join(' ');answer=ws.slice(n).join(' ');
  label='Ayetin devamını bul';
  const distractors=shuffle([...new Set(pool.map(r=>{
   const z=words(r.arabic);
   return z.slice(Math.max(1,Math.floor(z.length/2))).join(' ');
  }).filter(w=>w&&w!==answer))]).slice(0,3);
  opts=[answer,...distractors];
 }else if(type==='meaning'){
  prompt=target.arabic;answer=target.translation;label='Ayetin mealini seç';arabicOpts=false;
  const distractors=shuffle([...new Set(pool.map(r=>r.translation).filter(t=>t&&t!==answer))]).slice(0,3);
  opts=[answer,...distractors];
 }else if(type==='next'){
  const next=all.find(r=>r.ayah===target.ayah+1);
  if(!next)return null;
  prompt=target.arabic;answer=next.arabic;label='Sonraki ayeti bul';
  const distractors=shuffle([...new Set(pool.map(r=>r.arabic).filter(a=>a&&a!==answer&&a!==prompt))]).slice(0,3);
  opts=[answer,...distractors];
 }else return null;
 if(opts.length!==4||new Set(opts).size!==4)return null;
 const options=shuffle(opts).map((value,i)=>({id:'o'+i,text:value,correct:value===answer}));
 return {type,ayah:target.ayah,label,prompt,answer,options,arabicOpts};
}
function build(id){
 const all=registeredRows(id),pool=poolFor(id);
 if(all.length!==SURAHS[id].count||pool.length<7)return null;
 const result=[];
 for(const stage of STAGES){
  const n=stage.ayah||SURAHS[id].count;
  const target=all.find(r=>r.ayah===n);
  if(!target)return null;
  const q=question(stage.type,target,pool,all);
  if(!q)return null;
  result.push(q);
 }
 return result.length===TOTAL?result:null;
}
function currentSurah(){
 let id='';
 try{id=String(window.DEEN_QURAN?.ensureState?.()?.profile?.currentSurah||'')}catch(_){}
 return SURAHS[id]?id:'fatiha';
}
function visible(){return !!document.querySelector?.('#v5Sheet[open] #v5SheetBody .deen-atest')}
function draw(html){
 if(typeof sheet!=='function')return false;
 sheet('Ayet Testi',html);
 return true;
}
function selectionHtml(){
 const preferred=currentSurah();
 return '<div class="deen-atest" data-atest-phase="choose">'+
 '<div class="deen-atest-lead"><div class="deen-atest-tag">KUR’AN · OYNA</div><h3>Ayet Testi</h3>'+
 '<p>5 kısa soruyla ayetleri hatırla. Önce çalışmak istediğin sureyi seç.</p></div>'+
 Object.keys(SURAHS).sort((a,b)=>a===preferred?-1:b===preferred?1:0).map(id=>
 '<button class="deen-atest-pick" type="button" onclick="DEEN_QURAN_AYET_TEST.start(\''+id+'\')"><span><b>'+SURAHS[id].name+' Suresi</b><small>'+SURAHS[id].count+' ayet · 5 soruluk tur'+(id===preferred?' · Önerilen':'')+'</small></span><i>→</i></button>'
 ).join('')+
 '<p class="deen-atest-note">Sorular kayıtlı Arapça ayet metni ve Türkçe mealden oluşturulur. Bu sürümde ses ve sürükleme testi bulunmaz.</p></div>';
}
function displayArabic(q){
 if(Array.isArray(q.prompt)){
  return q.prompt.map(w=>w==='__BLANK__'?'<span class="deen-atest-blank">ــــــ</span>':esc(w)).join(' ');
 }
 return esc(q.prompt);
}
function bodyHtml(){
 if(!run)return selectionHtml();
 if(run.done)return resultHtml();
 const q=run.questions[run.index];
 if(!q)return'<div class="deen-atest">Soru bulunamadı.</div>';
 const checked=run.checked;
 const answerChoice=q.options.find(x=>x.correct);
 const choices=q.options.map(opt=>{
   const active=run.choice===opt.id;
   const klass=checked?(opt.correct?' good':active?' bad':''):active?' selected':'';
   return '<button type="button" class="deen-atest-choice'+(q.arabicOpts?' ar':'')+klass+'" aria-pressed="'+(active?'true':'false')+'"'+
   (checked?' disabled':'')+' onclick="DEEN_QURAN_AYET_TEST.select(\''+opt.id+'\')">'+esc(opt.text)+'</button>';
 }).join('');
 return '<div class="deen-atest" data-atest-phase="question">'+
 '<div class="deen-atest-progresshead"><span>'+esc(SURAHS[run.id].name)+' Suresi</span><b>'+(run.index+1)+' / '+TOTAL+'</b></div>'+
 '<div class="deen-atest-progress" aria-label="Test ilerlemesi">'+Array.from({length:TOTAL},(_,i)=>'<i class="'+(i<=run.index?'done':'')+'"></i>').join('')+'</div>'+
 '<div class="deen-atest-qcard"><div class="deen-atest-tag">AYET '+q.ayah+'</div><h3>'+esc(q.label)+'</h3>'+
 '<div class="deen-atest-ref">Kayıtlı ayet metni · '+esc(SURAHS[run.id].name)+'</div>'+
 '<div class="deen-atest-ar" lang="ar">'+displayArabic(q)+'</div>'+
 '<div class="deen-atest-options" role="group" aria-label="Cevap seçenekleri">'+choices+'</div>'+
 (checked?'<div class="deen-atest-answer '+(run.answerCorrect?'good':'bad')+'" role="status">'+
 (run.answerCorrect?'Doğru!':'Tekrar çalışılacak. Doğru cevap: '+esc(answerChoice?.text||q.answer))+'</div>':'')+
 (checked?
 '<button type="button" class="deen-atest-btn" onclick="DEEN_QURAN_AYET_TEST.next()">'+(run.index===TOTAL-1?'SONUCU GÖR':'SONRAKİ SORU →')+'</button>':
 '<button type="button" class="deen-atest-btn" onclick="DEEN_QURAN_AYET_TEST.check()"'+(!run.choice?' disabled':'')+'>KONTROL ET</button>')+
 '</div><p class="deen-atest-note">Ayet verileri kayıtlı kaynaklardan alınmıştır. Ezber ilerlemen bu testten otomatik değiştirilmez.</p></div>';
}
function resultHtml(){
 const pct=Math.round(run.correct/TOTAL*100);
 const missed=run.answers.filter(x=>!x.correct);
 return '<div class="deen-atest" data-atest-phase="result">'+
  '<div class="deen-atest-result"><div class="deen-atest-tag">'+esc(SURAHS[run.id].name)+' SURESİ · TAMAMLANDI</div>'+
  '<strong>%'+pct+'</strong><b>'+run.correct+' / '+TOTAL+' doğru</b>'+
  '<p>'+(missed.length?'Yanlış yaptığın ayetleri tekrar ederek güçlendirebilirsin.':'Bu turdaki tüm soruları doğru cevapladın!')+'</p></div>'+
  (missed.length?'<div class="deen-atest-misses"><b>Tekrar önerilen ayetler</b>'+
   [...new Set(missed.map(x=>x.ayah))].map(n=>'<span>'+esc(SURAHS[run.id].name)+' · Ayet '+n+'</span>').join('')+'</div>':'')+
  '<button type="button" class="deen-atest-btn" onclick="DEEN_QURAN_AYET_TEST.start(\''+run.id+'\')">YENİDEN TEST ET</button>'+
  '<button type="button" class="deen-atest-btn secondary" onclick="DEEN_QURAN_AYET_TEST.open()">SURE DEĞİŞTİR</button></div>';
}
function refresh(){
 if(!visible())return false;
 const body=document.querySelector('#v5SheetBody');
 if(!body)return false;
 body.innerHTML=bodyHtml();
 return true;
}
function open(){run=null;return draw(selectionHtml())}
function start(id){
 if(!SURAHS[id])return false;
 const questions=build(id);
 if(!questions){
  return draw('<div class="deen-atest"><div class="deen-atest-lead"><h3>Test henüz hazır değil</h3><p>'+esc(SURAHS[id].name)+' için gerekli ayet metni veya meal kayıtları eksik. Eksik içerik için soru uydurulmadı.</p></div><button class="deen-atest-btn secondary" onclick="DEEN_QURAN_AYET_TEST.open()">GERİ DÖN</button></div>');
 }
 run={id,questions,index:0,correct:0,choice:null,checked:false,answerCorrect:false,answers:[],done:false,saved:false};
 return draw(bodyHtml());
}
function select(id){
 if(!run||run.checked||run.done||!visible())return false;
 const q=run.questions[run.index];
 if(!q?.options.some(o=>o.id===id))return false;
 run.choice=id;refresh();return true;
}
function check(){
 if(!run||run.checked||run.done||!visible()||!run.choice)return false;
 const q=run.questions[run.index];
 const option=q?.options.find(x=>x.id===run.choice);
 if(!option)return false;
 run.checked=true;
 run.answerCorrect=!!option.correct;
 if(option.correct)run.correct++;
 run.answers.push({ayah:q.ayah,type:q.type,correct:!!option.correct});
 refresh();return run.answerCorrect;
}
function saveResult(){
 if(!run||run.saved)return false;
 run.saved=true;
 try{
  const journey=window.DEEN_QURAN?.ensureState?.()||state.quranJourney;
  if(journey && typeof journey==='object'){
   const previous=Array.isArray(journey.ayetTestHistory)?journey.ayetTestHistory:[];
   journey.ayetTestHistory=[...previous.slice(-9),{
    surahId:run.id,at:new Date().toISOString(),score:run.correct,total:TOTAL,
    wrongAyahs:run.answers.filter(a=>!a.correct).map(a=>a.ayah)
   }];
   if(typeof saveState==='function')saveState();
  }
 }catch(_){}
 return true;
}
function next(){
 if(!run||!run.checked||run.done||!visible())return false;
 if(run.index===TOTAL-1){
  run.done=true;saveResult();refresh();return true;
 }
 run.index++;run.choice=null;run.checked=false;run.answerCorrect=false;refresh();return true;
}
function snapshot(){
 return run?{build:BUILD,surah:run.id,index:run.index,total:TOTAL,score:run.correct,
 selected:run.choice,checked:run.checked,done:run.done,answers:run.answers.map(x=>({...x}))}:{build:BUILD,phase:'choose',total:TOTAL};
}
window.DEEN_QURAN_AYET_TEST={
 open,start,select,check,next,snapshot,
 available:id=>!!SURAHS[id]&&!!build(id),
 audit:()=>({build:BUILD,surahs:Object.keys(SURAHS),questions:TOTAL,
  fatihaReady:!!build('fatiha'),ikhlasReady:!!build('ikhlas'),entry:!!document.querySelector?.('.deen-atest-entry')})
};
document.documentElement.dataset.deenQuranAyetTest='v10974';
})();
