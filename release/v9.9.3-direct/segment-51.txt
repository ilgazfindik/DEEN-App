let player=null,currentId=null,status='idle';

function directUrl(file){
 return 'https://commons.wikimedia.org/wiki/Special:Redirect/file/'+encodeURIComponent(file).replace(/'/g,'%27');
}
function ensurePlayer(){
 if(player)return player;
 const A=window.Audio||window.HTMLAudioElement;
 if(!A)return null;
 player=new A();
 player.preload='metadata';
 player.addEventListener?.('play',()=>{status='playing';refresh()});
 player.addEventListener?.('pause',()=>{if(status!=='error')status='paused';refresh()});
 player.addEventListener?.('ended',()=>{status='done';currentId=null;refresh()});
 player.addEventListener?.('error',()=>{status='error';refresh()});
 return player;
}
function source(id){const x=SOURCES[id];return x?{...x,audioUrl:directUrl(x.file)}:null}
async function play(id){
 const s=SOURCES[id];if(!s)return false;
 const p=ensurePlayer();if(!p)return false;
 if(currentId!==id||!p.src){
   try{p.src=directUrl(s.file);p.currentTime=0}catch(_){}
   currentId=id;
 }
 try{
   const out=p.play();if(out&&typeof out.then==='function')await out;
   status='playing';refresh();DEEN_QURAN_ANALYTICS?.track?.('audio_play',{surahId:id,mode:'chapter'});return true;
 }catch(_){status='error';refresh();return false}
}
function pause(){
 const p=ensurePlayer();try{p?.pause()}catch(_){}
 status='paused';refresh();return true;
}
function stop(){
 const p=ensurePlayer();try{p?.pause();p.currentTime=0}catch(_){}
 currentId=null;status='idle';refresh();return true;
}
function toggle(id){
 const p=ensurePlayer();
 if(currentId===id&&p&&!p.paused)return pause();
 return play(id);
}
function currentJourneyId(){
 const h=document.querySelector?.('.deen-qj2-hero h2')?.textContent||'';
 for(const [id,s] of Object.entries(SOURCES))if(h.includes(s.name))return id;
 const profile=DEEN_QURAN?.ensureState?.()?.profile?.currentSurah;
 return SOURCES[profile]?profile:null;
}
function cardHtml(id){
 const s=SOURCES[id],playing=currentId===id&&status==='playing';
 return '<div class="deen-qchapter" data-qchapter="'+id+'">'+
  '<span class="deen-qchapter-copy"><b>Sureyi Dinle</b><small>'+s.name+' · Wikimedia Commons kıraati</small>'+
  '<span class="deen-qchapter-license">'+s.author+' · '+s.license+'</span></span>'+
  '<button class="deen-qchapter-play'+(playing?' playing':'')+'" type="button" onclick="DEEN_QURAN_CHAPTER_AUDIO.toggle(\''+id+'\')">'+(playing?'DURAKLAT':'▶ DİNLE')+'</button>'+
 '</div>';
}
function inject(){
 const hero=document.querySelector?.('.deen-qj2-hero');if(!hero||hero.querySelector('.deen-qchapter'))return false;
 const id=currentJourneyId();if(!id)return false;
 hero.insertAdjacentHTML('beforeend',cardHtml(id));return true;
}
function refresh(){
 const old=document.querySelector?.('.deen-qchapter');if(!old)return false;
 const id=old.getAttribute('data-qchapter');const h=document.createElement('div');h.innerHTML=cardHtml(id);old.replaceWith(h.firstElementChild);return true;
}
let timer=null;function schedule(){clearTimeout(timer);timer=setTimeout(inject,35)}
document.addEventListener?.('deen:navigating',stop);
document.addEventListener?.('deen:navigated',schedule);

if(typeof MutationObserver!=='undefined')new MutationObserver(schedule).observe(document.body,{childList:true,subtree:true});
schedule();

window.DEEN_QURAN_CHAPTER_AUDIO={play,pause,stop,toggle,source,sources:()=>JSON.parse(JSON.stringify(SOURCES)),status:()=>({status,currentId,available:Object.keys(SOURCES).length})};
