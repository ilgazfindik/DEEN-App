<script id="deen-v10938-quran-personal-plan-js">
(()=>{if(window.__deenV10938QuranPersonalPlan)return;window.__deenV10938QuranPersonalPlan=true;
const IDS=['fatiha','ikhlas','falaq','nas','kafirun','nasr','masad','fil','quraysh','kawthar','asr'];
function plan(){const q=DEEN_QURAN.ensureState();q.personalPlan=q.personalPlan&&typeof q.personalPlan==='object'?q.personalPlan:{active:true,dailyMinutes:5,weeklyDays:3,newAyahs:1,targetSurahs:['ikhlas','falaq','nas']};q.personalPlan.targetSurahs=Array.isArray(q.personalPlan.targetSurahs)?q.personalPlan.targetSurahs.filter(x=>IDS.includes(x)):['ikhlas'];return q.personalPlan}
function setPreset(k){const p=plan();if(k==='calm')Object.assign(p,{active:true,dailyMinutes:5,weeklyDays:3,newAyahs:1});if(k==='balanced')Object.assign(p,{active:true,dailyMinutes:10,weeklyDays:5,newAyahs:1});if(k==='intensive')Object.assign(p,{active:true,dailyMinutes:15,weeklyDays:7,newAyahs:2});DEEN_QURAN.setGoal?.('dailyMinutes',p.dailyMinutes);DEEN_QURAN.setGoal?.('weeklyDays',p.weeklyDays);DEEN_QURAN.setGoal?.('dailyAyahs',p.newAyahs);DEEN_QURAN.persistState?.();redraw();return p}
function nextNew(){const q=DEEN_QURAN.ensureState(),p=plan();for(const id of p.targetSurahs){const s=q.progress?.surahs?.[id];for(const [n,a] of Object.entries(s?.ayahs||{}))if(!a.completedAt&&DEEN_QURAN_CONTENT?.getAyah?.(id,Number(n)))return{kind:'new',id,n:Number(n),title:id+' · Ayet '+n}}return null}
function today(){const p=plan(),due=DEEN_QURAN_SRS?.dueItems?.(true)||[],tasks=[];if(due.length)tasks.push({kind:'review',count:due.length,title:due.length+' tekrar bekliyor'});const n=nextNew();if(n)tasks.push(n);tasks.push({kind:'words',title:'Sık Kur’an kelimelerinden kısa tekrar'});return{minutes:p.dailyMinutes,weeklyDays:p.weeklyDays,newAyahs:p.newAyahs,tasks:tasks.slice(0,3)}}
function start(){const t=today().tasks[0];if(!t)return false;if(t.kind==='review')return DEEN_QURAN_SRS?.openReview?.();if(t.kind==='new')return DEEN_QURAN.openSurah?.(t.id);if(t.kind==='words')return DEEN_QURAN_WORD_INTEL?.open?.();return false}
function html(){const p=plan(),t=today();return'<div class="deen-qplan"><div class="deen-qplan-card"><h3>Kişisel Kur’an Planın</h3><p>'+p.dailyMinutes+' dk/gün · '+p.weeklyDays+' gün/hafta · '+p.newAyahs+' yeni ayet</p><div class="deen-qplan-presets"><button class="deen-qplan-btn" onclick="DEEN_QURAN_PLAN.setPreset(\'calm\')">SAKİN</button><button class="deen-qplan-btn" onclick="DEEN_QURAN_PLAN.setPreset(\'balanced\')">DENGELİ</button><button class="deen-qplan-btn" onclick="DEEN_QURAN_PLAN.setPreset(\'intensive\')">YOĞUN</button></div></div>'+t.tasks.map(x=>'<div class="deen-qplan-task"><b>'+x.title+'</b>'+x.kind+'</div>').join('')+'<button class="deen-qplan-btn" onclick="DEEN_QURAN_PLAN.start()">BUGÜNKÜ PLANI BAŞLAT</button></div>'}
function redraw(){const x=document.querySelector?.('.deen-qplan');if(!x)return false;const h=document.createElement('div');h.innerHTML=html();x.replaceWith(h.firstElementChild);return true}
function open(){sheet('Kişisel Kur’an Planı',html());return true}
window.DEEN_QURAN_PLAN={open,plan,today,start,setPreset,nextNew};window.DEEN_QURAN.personalPlan=open;document.documentElement.dataset.deenQuranPlan='v10938';
})();
