function installLessonScrollSupport(){
 const lesson=document.getElementById('lesson');
 const sc=document.querySelector('#quizView .qarea');
 if(!lesson||!sc||lesson.dataset.scrollSupport==='1')return;
 lesson.dataset.scrollSupport='1';

 // Some preview/iframe hosts swallow the normal wheel chain.
 // Route wheel/trackpad movement directly into the lesson scroller.
 lesson.addEventListener('wheel',e=>{
   if(!lesson.classList.contains('active'))return;
   const multiplier=e.deltaMode===1?24:(e.deltaMode===2?Math.max(320,sc.clientHeight):1);
   const dy=e.deltaY*multiplier;
   if(Math.abs(dy)<0.5)return;
   const max=Math.max(0,sc.scrollHeight-sc.clientHeight);
   const next=Math.max(0,Math.min(max,sc.scrollTop+dy));
   if(next!==sc.scrollTop){
     sc.scrollTop=next;
     e.preventDefault();
   }
 },{passive:false});

 // Desktop keyboard fallback.
 document.addEventListener('keydown',e=>{
   if(!lesson.classList.contains('active'))return;
   if(['INPUT','TEXTAREA'].includes((document.activeElement&&document.activeElement.tagName)||''))return;
   let dy=0;
   if(e.key==='ArrowDown')dy=70;
   else if(e.key==='ArrowUp')dy=-70;
   else if(e.key==='PageDown'||e.key===' ')dy=Math.max(240,sc.clientHeight*.72);
   else if(e.key==='PageUp')dy=-Math.max(240,sc.clientHeight*.72);
   else if(e.key==='Home'){sc.scrollTop=0;e.preventDefault();return}
   else if(e.key==='End'){sc.scrollTop=sc.scrollHeight;e.preventDefault();return}
   if(dy){sc.scrollBy({top:dy,behavior:'smooth'});e.preventDefault()}
 });

 // Pointer drag fallback for desktop preview windows.
 let dragging=false,lastY=0,moved=false;
 sc.addEventListener('pointerdown',e=>{
   if(e.pointerType==='mouse' && e.button!==0)return;
   if(e.target.closest('button,input'))return;
   dragging=true;moved=false;lastY=e.clientY;
   try{sc.setPointerCapture(e.pointerId)}catch(_){}
 });
 sc.addEventListener('pointermove',e=>{
   if(!dragging)return;
   const d=lastY-e.clientY;
   if(Math.abs(d)>2)moved=true;
   sc.scrollTop+=d;lastY=e.clientY;
 });
 const stopDrag=()=>{dragging=false};
 sc.addEventListener('pointerup',stopDrag);
 sc.addEventListener('pointercancel',stopDrag);
}
