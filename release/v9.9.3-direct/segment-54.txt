
<style id="deen-v10947-fatiha-direct-inline-css">
.deen-fatiha-inline-hint{color:#789995!important}
.deen-fatiha-inline-hint.error{color:#dfa18b!important}
</style>
<script id="deen-v10947-fatiha-direct-inline-js">
(()=>{
if(window.__deenV10947FatihaDirectInline)return;window.__deenV10947FatihaDirectInline=true;

const BASE='./assets/quran/audio/fatiha/';
const FILES={
  1:'fatiha_1.mp3',2:'fatiha_2.mp3',3:'fatiha_3.mp3',4:'fatiha_4.mp3',
  5:'fatiha_5.mp3',6:'fatiha_6.mp3',7:'fatiha_7.mp3'
};
let player=null,currentAyah=null,status='idle';

function q(){return DEEN_QURAN?.ensureState?.()||null}
function visibleAyah(){
  const root=document.querySelector?.('.deen-qmulti-session');
  const text=root?.querySelector?.('.deen-ikhlas-card h2')?.textContent||root?.querySelector?.('.deen-ikhlas-progresshead b')?.textContent||'';
  const m=String(text).match(/Ayet\s*(\d+)/i);
  if(m)return Math.max(1,Math.min(7,Number(m[1])||1));
  return Math.max(1,Math.min(7,Number(q()?.profile?.currentAyah)||1));
}
function isFatiha(){
  const p=q()?.profile?.currentSurah;
  const head=document.querySelector?.('.deen-qmulti-session .deen-ikhlas-progresshead b')?.textContent||'';
  return p==='fatiha'||/Fâtiha/i.test(head);
}
function url(n){return BASE+FILES[n]+'?v=9252'}
function hint(msg,error=false){
  const h=document.querySelector?.('.deen-fatiha-inline-hint');
  if(!h)return false;
  h.textContent=msg;
  h.classList.toggle('error',!!error);
  return true;
}
function ensurePlayer(){
  if(player)return player;
  if(typeof Audio!=='function')return null;
  player=new Audio();
  player.preload='auto';
  player.addEventListener('play',()=>{status='playing';syncButton()});
  player.addEventListener('pause',()=>{if(status!=='error'&&status!=='ended')status='paused';syncButton()});
  player.addEventListener('ended',()=>{status='ended';syncButton();hint('Ayet sesi tamamlandı. Tekrar dinlemek için hoparlöre dokun.')});
  player.addEventListener('error',()=>{status='error';syncButton();hint('Ayet sesi yüklenemedi. Sayfayı yenileyip tekrar dene.',true)});
  return player;
}
function syncButton(){
  const b=document.querySelector?.('.deen-fatiha-inline-speaker');
  if(!b)return false;
  const n=visibleAyah();
  const playing=status==='playing'&&currentAyah===n&&player&&!player.paused;
  b.classList.toggle('playing',playing);
  b.textContent=playing?'Ⅱ':'🔊';
  b.setAttribute('aria-label',playing?'Ayet sesini duraklat':'Ayet '+n+' sesini dinle');
  b.setAttribute('data-ayah-audio',String(n));
  return true;
}
async function play(n){
  n=Math.max(1,Math.min(7,Number(n)||1));
  const p=ensurePlayer();
  if(!p){hint('Bu tarayıcı ses oynatmayı desteklemiyor.',true);return false}
  if(currentAyah!==n||!p.src){
    try{p.pause()}catch(_){}
    currentAyah=n;
    p.src=url(n);
    try{p.currentTime=0}catch(_){}
  }
  hint('Ayet '+n+' dinleniyor…');
  try{
    const out=p.play();
    if(out&&typeof out.then==='function')await out;
    status='playing';syncButton();
    DEEN_QURAN_ANALYTICS?.track?.('audio_play',{surahId:'fatiha',mode:'ayah_local'});
    return true;
  }catch(e){
    status='error';syncButton();hint('Ayet sesi açılamadı: '+String(e?.message||e),true);
    return false;
  }
}
function pause(){
  if(!player)return false;
  try{player.pause()}catch(_){}
  status='paused';syncButton();hint('Ayet sesi duraklatıldı.');
  return true;
}
async function toggle(){
  const n=visibleAyah();
  if(player&&currentAyah===n&&!player.paused&&status==='playing')return pause();
  return play(n);
}
function stop(){
  if(player){try{player.pause();player.currentTime=0}catch(_){}}
  currentAyah=null;status='idle';syncButton();return true;
}
function decorate(){
  const root=document.querySelector?.('.deen-qmulti-session');
  if(!root||!isFatiha())return false;
  root.setAttribute('data-fatiha-inline-audio','1');
  const card=root.querySelector('.deen-ikhlas-card');
  if(!card)return false;
  let button=card.querySelector('.deen-fatiha-inline-speaker');
  if(!button){
    const h2=card.querySelector('h2');
    if(h2){
      let row=card.querySelector('.deen-fatiha-inline-row');
      if(!row){
        row=document.createElement('div');row.className='deen-fatiha-inline-row';
        h2.parentNode.insertBefore(row,h2);row.appendChild(h2);
      }
      button=document.createElement('button');
      button.type='button';button.className='deen-fatiha-inline-speaker';
      button.setAttribute('onclick','DEEN_QURAN_FATIHA_INLINE_AUDIO.toggle()');
      row.appendChild(button);
    }
  }
  let h=card.querySelector('.deen-fatiha-inline-hint');
  if(!h){
    h=document.createElement('p');h.className='deen-fatiha-inline-hint';
    const row=card.querySelector('.deen-fatiha-inline-row');
    row?.insertAdjacentElement('afterend',h);
  }
  if(h&&!h.classList.contains('error'))h.textContent='Hoparlöre dokun: Ayet '+visibleAyah()+' kaydını dinle.';
  const n=visibleAyah();
  if(currentAyah&&currentAyah!==n&&status==='playing')stop();
  syncButton();return true;
}
function audit(){
  return{
    ok:Object.keys(FILES).length===7,
    files:Object.entries(FILES).map(([n,file])=>({ayah:Number(n),file,url:BASE+file})),
    wikimediaKept:!!window.DEEN_QURAN_CHAPTER_AUDIO?.source?.('fatiha'),
    registryIndependent:true
  };
}
let timer=null;
function schedule(){clearTimeout(timer);timer=setTimeout(decorate,25)}
document.addEventListener?.('click',e=>{if(e.target?.closest?.('#v5Sheet'))schedule()},true);
document.addEventListener?.('deen:navigating',stop);
document.addEventListener?.('deen:navigated',schedule);
document.addEventListener?.('close',e=>{if(e.target?.id==='v5Sheet')stop()},true);
if(typeof MutationObserver!=='undefined')new MutationObserver(schedule).observe(document.body,{childList:true,subtree:true});
setInterval(()=>{decorate();syncButton()},300);
schedule();

window.DEEN_QURAN_FATIHA_INLINE_AUDIO={toggle,play,pause,stop,decorate,syncButton,ayah:visibleAyah,audit,state:()=>({status,currentAyah,src:player?.src||null})};
document.documentElement.dataset.deenFatihaInlineAudio='v10947';
})();
</script>
