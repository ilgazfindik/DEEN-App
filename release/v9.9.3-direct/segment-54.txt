<script id="deen-v10964-quran-challenge-js">
(()=>{

};

const qs=()=>{try{return DEEN_QURAN?.ensureState?.()||state?.quranJourney||{}}catch(_){return state?.quranJourney||{}}};
function persist(){try{DEEN_QURAN?.persistState?.()}catch(_){}}
function store(){
 const q=qs();q.questionArenaMastery=q.questionArenaMastery&&typeof q.questionArenaMastery==='object'?q.questionArenaMastery:{version:1,surahs:{},challenge:{plays:0,bestAccuracy:0,bestStreak:0}};

function masteryScore(id){const s=store().surahs[id],attempts=s.correct+s.wrong;if(!attempts)return 50;const accuracy=s.correct/attempts;const confidence=Math.min(1,attempts/10);return Math.round((.5*(1-confidence)+accuracy*confidence)*100)}

}

 Object.entries(LEVELS).map(([id,x])=>'<button class="deen-qchallenge-level" type="button" onclick="DEEN_QURAN_CHALLENGE.start(\''+id+'\')"><b>'+x.name+'<em>'+x.tag+'</em></b><small>'+x.types.map(t=>({surah:'Hangi Sure',ayah_number:'Ayet No',meaning:'Meal',fill_blank:'Boşluk',continuation:'Devamı',audio:'Sesi Bul',next_ayah:'Sonraki Ayet'})[t]).join(' · ')+'</small></button>').join('')+
 '</div>'}

 const root=document.querySelector('.deen-qarena');if(!root)return false;
 const score=root.querySelector('.deen-qarena-score');if(score&&C.active)score.textContent='✓ '+C.correct+' · 🔥 '+C.streak;

 const current=Math.min(C.limit,C.answered+1),pct=Math.round((C.answered/C.limit)*100);

 if(!LEVELS[level])level='medium';
 C={active:true,level,limit:10,answered:0,correct:0,wrong:0,streak:0,bestStreak:0,history:[],startedAt:new Date().toISOString(),finishedAt:null};

 const root=store();let rows=IDS.map(id=>({id,name:META[id],score:masteryScore(id),attempts:root.surahs[id].correct+root.surahs[id].wrong})).filter(x=>x.attempts>0);
 if(!rows.length)rows=[{id:'fatiha',name:META.fatiha,score:50,attempts:0}];rows.sort((a,b)=>b.score-a.score||b.attempts-a.attempts);return rows[0]
}

 const label=acc>=90?'Çok güçlü':acc>=70?'İyi gidiyor':acc>=50?'Gelişiyor':'Tekrar gerekli';

 '<div class="deen-qchallenge-metrics"><div><strong>'+C.correct+'/10</strong><small>DOĞRU</small></div><div><strong>'+C.bestStreak+'</strong><small>EN İYİ SERİ</small></div><div><strong>'+top.score+'</strong><small>'+top.name.toUpperCase()+'</small></div></div>'+
 '<div class="deen-qchallenge-actions"><button class="deen-qchallenge-btn" onclick="DEEN_QURAN_CHALLENGE.start(\''+C.level+'\')">TEKRAR OYNA</button><button class="deen-qchallenge-btn secondary" onclick="DEEN_QURAN_CHALLENGE.openMastery()">USTALIKLARIM</button></div></div>'
}
function finish(){
 C.active=false;C.finishedAt=new Date().toISOString();const acc=C.answered?Math.round(C.correct/C.answered*100):0;const root=store();root.challenge.plays=Math.max(0,Number(root.challenge.plays)||0)+1;root.challenge.bestAccuracy=Math.max(Number(root.challenge.bestAccuracy)||0,acc);root.challenge.bestStreak=Math.max(Number(root.challenge.bestStreak)||0,C.bestStreak);root.challenge.lastLevel=C.level;root.challenge.lastAt=C.finishedAt;persist();sheet('Challenge Sonucu',resultHtml());return true
}

 const root=store();const rows=IDS.map(id=>({id,name:META[id],score:masteryScore(id),c:root.surahs[id].correct,w:root.surahs[id].wrong})).sort((a,b)=>b.score-a.score||(b.c+b.w)-(a.c+a.w));

 return true

function install(){
 const a=arena();if(!a||a.__challengeWrapped)return false;
 const baseChoose=a.choose.bind(a),baseNext=a.next.bind(a),baseOpen=a.open.bind(a),baseSetType=a.setType.bind(a),baseSetScope=a.setScope.bind(a);
 a.open=function(){const r=baseOpen(...arguments);setTimeout(()=>{decorate();injectHub()},0);return r};
 a.choose=function(value){
   if(!C.active)return baseChoose(value);
   const before=a.state?.();const q=before?.q;if(!q||before?.answered)return false;
   const ok=baseChoose(value);C.answered++;if(ok){C.correct++;C.streak++;C.bestStreak=Math.max(C.bestStreak,C.streak)}else{C.wrong++;C.streak=0}
   const score=record(q.target.id,!!ok,C.streak);C.history.push({type:q.type,surahId:q.target.id,ayah:q.target.ayah,ok:!!ok,mastery:score});
   setTimeout(decorate,0);return ok
 };
 a.next=function(){
   if(!C.active)return baseNext();
   if(C.answered>=C.limit)return finish();
   const t=levelType();const r=baseSetType(t);setTimeout(decorate,0);return r
 };
 a.setType=function(t){if(C.active)return false;return baseSetType(t)};
 a.setScope=function(s){if(C.active)return false;return baseSetScope(s)};
 a.__challengeWrapped=true;return true
}

setTimeout(()=>{install();injectHub()},0);document.documentElement.dataset.deenQuranChallenge='v10964';
})();
