<script id="deen-v10949-ikhlas-inline-audio-js">
(()=>{
if(window.__deenV10949IkhlasInlineAudio)return;window.__deenV10949IkhlasInlineAudio=true;

const BASE='./assets/quran/audio/ikhlas/';
const FILES={1:'ihlas_1.mp3',2:'ihlas_2.mp3',3:'ihlas_3.mp3',4:'ihlas_4.mp3'};
let player=null,currentAyah=null,status='idle',rafId=0;

function q(){return DEEN_QURAN?.ensureState?.()||null}
function visibleAyah(){
  const root=document.querySelector?.('.deen-ikhlas-session');
  const text=root?.querySelector?.('.deen-ikhlas-card h2')?.textContent||
             root?.querySelector?.('.deen-ikhlas-progresshead b')?.textContent||'';
  const m=String(text).match(/Ayet\s*(\d+)/i);
  if(m)return Math.max(1,Math.min(4,Number(m[1])||1));
  return Math.max(1,Math.min(4,Number(q()?.profile?.currentAyah)||1));
}
function isIkhlas(){
  const p=q()?.profile?.currentSurah;
  const head=document.querySelector?.('.deen-ikhlas-session .deen-ikhlas-progresshead b')?.textContent||'';
  return p==='ikhlas'||/İhlâs/i.test(head);
}
function url(n){return BASE+FILES[n]+'?v=9254'}
function hint(msg,error=false){
  const h=document.querySelector?.('.deen-ikhlas-inline-hint');
  if(!h)return false;
  h.textContent=msg;
  h.classList.toggle('error',!!error);
  return true;
}
function cleanWord(w){
  return String(w||'')
    .replace(/[\u064B-\u065F\u0670\u06D6-\u06ED]/g,'')
    .replace(/[^\u0621-\u064A]/g,'');
}
function prepareSyncedText(){
  const root=document.querySelector?.('.deen-ikhlas-session');
  const ar=root?.querySelector?.('.deen-ikhlas-card .deen-ikhlas-arabic');
  if(!ar)return null;
  const n=visibleAyah();
  if(ar.dataset.ikhlasSync==='1'&&Number(ar.dataset.syncAyah)===n)return ar;
  const raw=String(ar.textContent||'').trim();
  if(!raw)return null;
  const list=raw.split(/\s+/).filter(Boolean);
  ar.innerHTML=list.map((w,i)=>
    '<span class="deen-ikhlas-sync-word" data-sync-index="'+i+'" data-sync-weight="'+
    Math.max(1,cleanWord(w).length)+'">'+w+'</span>'
  ).join(' ')+'<div class="deen-ikhlas-sync-track" aria-hidden="true"><i></i></div>';
  ar.dataset.ikhlasSync='1';
  ar.dataset.syncAyah=String(n);
  return ar;
}
function resetSyncedText(clearProgress=true){
  const ar=document.querySelector?.('.deen-ikhlas-session .deen-ikhlas-card .deen-ikhlas-arabic[data-ikhlas-sync="1"]');
  if(!ar)return false;
  for(const w of ar.querySelectorAll('.deen-ikhlas-sync-word'))w.classList.remove('past','current');
  if(clearProgress){
    const bar=ar.querySelector('.deen-ikhlas-sync-track i');
    if(bar)bar.style.width='0%';
  }
  return true;
}
function syncText(){
  const ar=prepareSyncedText();
  if(!ar||!player||currentAyah!==visibleAyah())return false;
  const spans=[...ar.querySelectorAll('.deen-ikhlas-sync-word')];
  const bar=ar.querySelector('.deen-ikhlas-sync-track i');
  const dur=Number(player.duration),t=Number(player.currentTime)||0;
  if(!Number.isFinite(dur)||dur<=0||!spans.length)return false;

  const progress=Math.max(0,Math.min(1,t/dur));
  if(bar)bar.style.width=(progress*100).toFixed(2)+'%';

  if(status==='ended'){
    spans.forEach(w=>{w.classList.add('past');w.classList.remove('current')});
    return true;
  }

  const lead=Math.min(.16,dur*.07),tail=Math.min(.12,dur*.05);
  const usable=Math.max(.1,dur-lead-tail);
  const p=Math.max(0,Math.min(.999,(t-lead)/usable));
  const weights=spans.map(w=>Math.max(1,Number(w.dataset.syncWeight)||1));
  const total=weights.reduce((a,b)=>a+b,0);
  let acc=0,idx=0;
  for(let i=0;i<weights.length;i++){
    acc+=weights[i]/total;
    if(p<acc){idx=i;break}
  }
  spans.forEach((w,i)=>{
    w.classList.toggle('past',i<idx);
    w.classList.toggle('current',i===idx&&t>=lead);
  });
  return true;
}
function stopRaf(){
  if(rafId&&typeof cancelAnimationFrame==='function')cancelAnimationFrame(rafId);
  rafId=0;
}
function startRaf(){
  stopRaf();
  const step=()=>{
    syncText();
    if(status==='playing'&&typeof requestAnimationFrame==='function')rafId=requestAnimationFrame(step);
  };
  if(typeof requestAnimationFrame==='function')rafId=requestAnimationFrame(step);
}
function ensurePlayer(){
  if(player)return player;
  if(typeof Audio!=='function')return null;
  player=new Audio();
  player.preload='auto';
  player.addEventListener('play',()=>{
    status='playing';syncButton();prepareSyncedText();startRaf();syncText();
  });
  player.addEventListener('pause',()=>{
    if(status!=='error'&&status!=='ended')status='paused';
    stopRaf();syncButton();syncText();
  });
  player.addEventListener('ended',()=>{
    status='ended';stopRaf();syncButton();syncText();
    hint('Ayet sesi tamamlandı. Tekrar dinlemek için hoparlöre dokun.');
  });
  player.addEventListener('error',()=>{
    status='error';stopRaf();syncButton();
    hint('Ayet sesi yüklenemedi. Sayfayı yenileyip tekrar dene.',true);
  });
  player.addEventListener('loadedmetadata',syncText);
  player.addEventListener('timeupdate',syncText);
  return player;
}
function syncButton(){
  const b=document.querySelector?.('.deen-ikhlas-inline-speaker');
  if(!b)return false;
  const n=visibleAyah();
  const playing=status==='playing'&&currentAyah===n&&player&&!player.paused;
  b.classList.toggle('playing',playing);
  b.textContent=playing?'Ⅱ':'🔊';
  b.setAttribute('aria-label',playing?'Ayet sesini duraklat':'Ayet '+n+' sesini dinle');
  b.setAttribute('data-ayah-audio',String(n));
  return true;
}
async function play(n){
  n=Math.max(1,Math.min(4,Number(n)||1));
  const p=ensurePlayer();
  if(!p){
    hint('Bu tarayıcı ses oynatmayı desteklemiyor.',true);
    return false;
  }
  if(currentAyah!==n||!p.src){
    try{p.pause()}catch(_){}
    currentAyah=n;
    resetSyncedText(true);
    prepareSyncedText();
    p.src=url(n);
    try{p.currentTime=0}catch(_){}
  }
  hint('Ayet '+n+' dinleniyor…');
  try{
    const out=p.play();
    if(out&&typeof out.then==='function')await out;
    status='playing';syncButton();startRaf();syncText();
    DEEN_QURAN_ANALYTICS?.track?.('audio_play',{surahId:'ikhlas',mode:'ayah_local'});
    return true;
  }catch(e){
    status='error';syncButton();
    hint('Ayet sesi açılamadı: '+String(e?.message||e),true);
    return false;
  }
}
function pause(){
  if(!player)return false;
  try{player.pause()}catch(_){}
  status='paused';stopRaf();syncButton();syncText();
  hint('Ayet sesi duraklatıldı.');
  return true;
}
async function toggle(){
  const n=visibleAyah();
  if(player&&currentAyah===n&&!player.paused&&status==='playing')return pause();
  return play(n);
}
function stop(){
  if(player){
    try{player.pause();player.currentTime=0}catch(_){}
  }
  currentAyah=null;status='idle';stopRaf();resetSyncedText(true);syncButton();
  return true;
}
function decorate(){
  const root=document.querySelector?.('.deen-ikhlas-session');
  if(!root||!isIkhlas())return false;

  root.setAttribute('data-ikhlas-inline-audio','1');
  const card=root.querySelector('.deen-ikhlas-card');
  if(!card)return false;

  const stage=card.querySelector('.deen-ikhlas-kicker')?.textContent||'';
  if(!/DİNLE/i.test(stage))return false;

  let button=card.querySelector('.deen-ikhlas-inline-speaker');
  if(!button){
    const h2=card.querySelector('h2');
    if(h2){
      let row=card.querySelector('.deen-ikhlas-audio-row');
      if(!row){
        row=document.createElement('div');
        row.className='deen-ikhlas-audio-row';
        h2.parentNode.insertBefore(row,h2);
        row.appendChild(h2);
      }
      button=document.createElement('button');
      button.type='button';
      button.className='deen-ikhlas-inline-speaker';
      button.setAttribute('onclick','DEEN_QURAN_IKHLAS_INLINE_AUDIO.toggle()');
      row.appendChild(button);
    }
  }

  let h=card.querySelector('.deen-ikhlas-inline-hint');
  if(!h){
    h=document.createElement('p');
    h.className='deen-ikhlas-inline-hint';
    const row=card.querySelector('.deen-ikhlas-audio-row');
    row?.insertAdjacentElement('afterend',h);
  }
  if(h&&!h.classList.contains('error')){
    h.textContent='Hoparlöre dokun: Ayet '+visibleAyah()+' kaydını dinle.';
  }

  prepareSyncedText();
  syncText();

  const n=visibleAyah();
  if(currentAyah&&currentAyah!==n&&status==='playing')stop();
  syncButton();
  return true;
}
function audit(){
  return{
    ok:Object.keys(FILES).length===4,
    files:Object.entries(FILES).map(([n,file])=>({
      ayah:Number(n),file,url:BASE+file
    })),
    wikimediaKept:!!window.DEEN_QURAN_CHAPTER_AUDIO?.source?.('ikhlas'),
    registryIndependent:true,
    wordSync:true
  };
}

let timer=null;
function schedule(){clearTimeout(timer);timer=setTimeout(decorate,25)}
document.addEventListener?.('click',e=>{
  if(e.target?.closest?.('#v5Sheet'))schedule();
},true);
document.addEventListener?.('deen:navigating',stop);
document.addEventListener?.('deen:navigated',schedule);
document.addEventListener?.('close',e=>{
  if(e.target?.id==='v5Sheet')stop();
},true);
if(typeof MutationObserver!=='undefined'){
  new MutationObserver(schedule).observe(document.body,{childList:true,subtree:true});
}
setInterval(()=>{decorate();syncButton()},300);
schedule();

window.DEEN_QURAN_IKHLAS_INLINE_AUDIO={
  toggle,play,pause,stop,decorate,syncButton,syncText,prepareSyncedText,
  ayah:visibleAyah,audit,
  state:()=>({
    status,currentAyah,src:player?.src||null,currentTime:player?.currentTime||0,
    duration:Number.isFinite(player?.duration)?player.duration:null
  })
};
document.documentElement.dataset.deenIkhlasInlineAudio='v10949';
})();
</script>
