<script id="deen-v1077-profile-avatar-assets-js">
(()=>{
 if(window.__deenV1077ProfileAvatarAssets)return;
 window.__deenV1077ProfileAvatarAssets=true;

 const VERSION='10.7.7';
 const ASSETS={
   lion:'./assets/companions/lion/lion_avatar.png',
   wolf:'./assets/companions/wolf/wolf_avatar.png',
   owl:'./assets/companions/owl/owl_avatar.png',
   turtle:'./assets/companions/turtle/turtle_avatar.png',
   eagle:'./assets/companions/eagle/eagle_avatar.png',
   cat:'./assets/companions/cat/cat_avatar.png'
 };
 const LABELS={
   lion:'Aslan',
   wolf:'Kurt',
   owl:'Baykuş',
   turtle:'Kaplumbağa',
   eagle:'Kartal',
   cat:'Kedi'
 };

 function selected(){
   try{
     const id=String(state?.profile?.companion||'').trim().toLowerCase();
     return ASSETS[id]?id:'';
   }catch(_){return ''}
 }

 function apply(){
   const root=document.getElementById('profileScreenBody');
   const avatar=root?.querySelector('.v5-profilehead .v5-avatar.large');
   const id=selected();
   if(!avatar||!id)return false;

   const wanted=ASSETS[id];
   const existing=avatar.querySelector('.v1077-profile-mascot');
   if(existing&&existing.dataset.animal===id&&existing.getAttribute('src')?.includes(wanted)){
     return true;
   }

   avatar.classList.add('v994-companion-avatar','v1077-profile-avatar');
   avatar.innerHTML='';

   const img=document.createElement('img');
   img.className='v1077-profile-mascot';
   img.dataset.animal=id;
   img.src=wanted+'?v=9178profileavatarrestore1';
   img.alt='';
   img.setAttribute('aria-hidden','true');
   avatar.appendChild(img);

   const label=LABELS[id]||id;
   avatar.title=label;
   avatar.setAttribute('aria-label','Yol arkadaşın '+label+'. Değiştirmek için dokun.');
   if(typeof window.DEEN_COMPANION?.open==='function'){
     avatar.onclick=()=>window.DEEN_COMPANION.open('profile');
   }

   avatar.dataset.mascotAsset='original-avatar';
   avatar.dataset.animal=id;
   document.documentElement.dataset.deenProfileMascot=id;
   return true;
 }

 function schedule(){
   requestAnimationFrame(apply);
   setTimeout(apply,40);
   setTimeout(apply,120);
 }

 document.addEventListener('deen:profile-rendered',schedule);
 document.addEventListener('deen:ui-updated',schedule);
 document.addEventListener('deen:navigated',schedule);

 const body=document.getElementById('profileScreenBody');
 if(body)new MutationObserver(schedule).observe(body,{childList:true,subtree:true});

 const baseSelect=window.DEEN_COMPANION?.select;
 if(typeof baseSelect==='function'&&!baseSelect.__deenV1077Wrapped){
   const wrapped=function(){
     const out=baseSelect.apply(this,arguments);
     schedule();
     return out;
   };
   wrapped.__deenV1077Wrapped=true;
   window.DEEN_COMPANION.select=wrapped;
 }

 window.DEEN_PROFILE_AVATAR_ASSETS={
   version:VERSION,
   assets:{...ASSETS},
   apply,
   check:()=>({
     selected:selected(),
     src:document.querySelector('#profileScreenBody .v1077-profile-mascot')?.getAttribute('src')||null,
     rendered:document.querySelector('#profileScreenBody .v1077-profile-mascot')?.dataset?.animal||null
   })
 };

 schedule();
})();
</script>
