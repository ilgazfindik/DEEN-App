<script id="deen-v1089-multilingual-typography-js">
(()=>{
  if(window.__deenV1089Typography)return;
  window.__deenV1089Typography=true;

  const RTL=new Set(['ar','fa','ur','he']);
  const normalize=locale=>String(locale||'tr').trim().toLowerCase().replace('_','-');

  function setLocale(locale){
    const value=normalize(locale);
    const base=value.split('-')[0]||'tr';
    document.documentElement.lang=value;
    document.documentElement.dir=RTL.has(base)?'rtl':'ltr';
    document.documentElement.dataset.deenLocale=value;
    return {locale:value,dir:document.documentElement.dir};
  }

  const current=normalize(document.documentElement.lang||'tr');
  if(!document.documentElement.dir){
    document.documentElement.dir=RTL.has(current.split('-')[0])?'rtl':'ltr';
  }
  document.documentElement.dataset.deenTypography='multilingual-v1';

  if(document.fonts?.ready){
    document.fonts.ready.then(()=>{
      document.documentElement.dataset.deenFontsReady='1';
    }).catch(()=>{});
  }

  window.DEEN_TYPOGRAPHY={
    version:'10.8.9',
    build:'9190multitype1',
    primary:'Nunito Sans',
    fallback:'Noto Sans',
    arabic:'Noto Sans Arabic',
    setLocale,
    check:()=>({
      locale:document.documentElement.lang,
      dir:document.documentElement.dir,
      font:getComputedStyle(document.body).fontFamily,
      ready:document.documentElement.dataset.deenFontsReady||'0',
      css:!!document.getElementById('deen-v1089-multilingual-typography-css')
    })
  };
})();
</script>
