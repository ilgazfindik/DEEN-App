<script id="deen-v1091-typography-freeze-js">
(()=>{
  if(window.__deenV1091TypographyFreeze)return;
  window.__deenV1091TypographyFreeze=true;

  const IDS=[
    'deen-v1089-multilingual-typography-css',
    'deen-v1090-readability-guard-css',
    'deen-v1091-typography-freeze-css',
    'deen-v1092-luxury-neon-home-css'
  ];

  function enforce(){
    const head=document.head;
    if(!head)return false;

    const nodes=IDS.map(id=>document.getElementById(id)).filter(Boolean);
    if(!nodes.length)return false;

    // Only reorder when necessary. Never keep observing <head>;
    // the previous observer could create an append -> mutation -> append loop on mobile.
    let lastIndex=-1;
    let inOrder=true;
    const children=[...head.children];
    for(const node of nodes){
      const idx=children.indexOf(node);
      if(idx<=lastIndex){inOrder=false;break;}
      lastIndex=idx;
    }

    const freeze=nodes[nodes.length-1];
    const alreadyLast=freeze===head.lastElementChild;
    if(!inOrder||!alreadyLast){
      const frag=document.createDocumentFragment();
      for(const node of nodes)frag.appendChild(node);
      head.appendChild(frag);
    }

    document.documentElement.dataset.deenTypographyLocked='1';
    return true;
  }

  // One initial pass is enough because these styles ship in the final release segment.
  // A second delayed pass covers any late parser/layout timing without a permanent observer.
  enforce();
  if(document.readyState==='loading'){
    document.addEventListener('DOMContentLoaded',enforce,{once:true});
  }
  setTimeout(enforce,120);

  window.DEEN_TYPOGRAPHY_LOCK={
    version:'10.9.6',
    build:'9200metacascadefix1',
    family:'Nunito Sans',
    neonPreserved:true,
    multilingual:true,
    observer:false,
    enforce,
    check:()=>({
      locked:document.documentElement.dataset.deenTypographyLocked==='1',
      font:getComputedStyle(document.body).fontFamily,
      typography:!!document.getElementById('deen-v1089-multilingual-typography-css'),
      readability:!!document.getElementById('deen-v1090-readability-guard-css'),
      freeze:!!document.getElementById('deen-v1091-typography-freeze-css'),
      observer:false
    })
  };
})();
