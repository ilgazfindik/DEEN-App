<script id="deen-v1075-t-dom-autocorrect-js">
(()=>{
 if(window.__deenV1075TDomAutoCorrect)return;
 window.__deenV1075TDomAutoCorrect=true;
 const n=v=>String(v??'').trim().toLocaleLowerCase('tr-TR').replace(/\s+/g,' ');
 const qNow=()=>{try{return session?.questions?.[Number(session?.index)||0]||null}catch(_){return null}};
 const active=()=>!!document.getElementById('lesson')?.classList.contains('active');
 const feedback=()=>!!document.getElementById('feedback')?.classList.contains('show');
 const hit=(sel,val,get)=>[...document.querySelectorAll(sel)].find(x=>!x.disabled&&n(get?get(x):(x.dataset?.value??x.textContent))===n(val));

 function matching(q){
   if(q?.activity_type!=='matching')return false;
   const area=document.getElementById('answerArea');
   if(!area?.querySelector('.v1017-card'))return false;
   try{if(session?.__deenMatchingState){session.__deenMatchingState.errors=0;session.__deenMatchingState.selected=null}}catch(_){}
   area.querySelector('.v1017-card.selected:not(.matched)')?.click();
   const ids=[...new Set([...area.querySelectorAll('.v1017-card[data-side="left"][data-pair]')].filter(x=>!x.classList.contains('matched')).map(x=>x.dataset.pair))];
   for(const id of ids){
     area.querySelector('.v1017-card[data-side="left"][data-pair="'+id+'"]:not(.matched)')?.click();
     area.querySelector('.v1017-card[data-side="right"][data-pair="'+id+'"]:not(.matched)')?.click();
   }
   return ids.length>0;
 }

 function sequence(q){
   if(!Array.isArray(q?.correct_order)||!document.querySelector('#answerArea .sequence-bank'))return false;
   for(const val of q.correct_order){
     const b=hit('#answerArea .sequence-bank .seq-chip:not(:disabled)',val,x=>String(x.textContent||'').replace(/^\s*\d+\.\s*/,''));
     b?.click();
   }
   const check=document.querySelector('#answerArea > .btn,#answerArea .template-action');
   if(check&&!check.disabled)check.click();
   return true;
 }

 function choice(q){
   if(q?.correct_answer==null)return false;
   let b=hit('#answerArea .fill-choice,#answerArea .v1039-fill-chip,#answerArea .v1040-fill-chip',q.correct_answer,x=>x.dataset?.value??x.textContent);
   if(!b)b=hit('#answerArea .option',q.correct_answer,x=>x.querySelector('.option-label')?.textContent??x.textContent);
   if(!b)return false;
   b.click();
   const c=document.querySelector('#answerArea .template-action,#answerArea .v1039-fill-check,#answerArea .v1040-fill-check');
   if(c&&!c.disabled)c.click();
   return true;
 }

 function textAnswer(q){
   if(q?.correct_answer==null)return false;
   const i=document.querySelector('#answerArea .fill-input,#answerArea .answer-input,#answerArea input[type="text"]');
   if(!i||i.disabled)return false;
   i.value=String(q.correct_answer);
   i.dispatchEvent(new Event('input',{bubbles:true}));
   i.dispatchEvent(new Event('change',{bubbles:true}));
   const c=document.querySelector('#answerArea .template-action,#answerArea .btn');
   if(c&&!c.disabled)c.click();
   return true;
 }

 function run(){
   if(!active()||feedback())return false;
   const q=qNow(); if(!q)return false;
   try{if(session?.answeredIndex===session?.index||session?.transitioning||session?.committed)return false}catch(_){}
   const ok=matching(q)||sequence(q)||choice(q)||textAnswer(q);
   document.documentElement.dataset.deenTShortcut=ok?'correct':'failed';
   return ok;
 }

 window.addEventListener('keydown',e=>{
   const isT=e.code==='KeyT'||String(e.key||'').toLowerCase()==='t'||e.keyCode===84;
   if(!isT||e.repeat||e.ctrlKey||e.metaKey||e.altKey||!active())return;
   e.preventDefault();
   e.stopPropagation();
   run();
 },true);

 window.DEEN_T_AUTOCORRECT={version:'10.7.5',trigger:run};
 document.documentElement.dataset.deenTShortcut='ready';
})();
</script>
