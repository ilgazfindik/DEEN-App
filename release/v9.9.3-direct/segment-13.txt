    if(p.challenge||p.key==='mastery_rehearsal'||p.key==='mixed_apply'){
      // Accessible opening, harder ending.
      return copy.sort((a,b)=>difficultyRank(a)-difficultyRank(b));

    if(p.key==='active_recall'||p.key==='retrieve'||p.key==='listen_retrieve'){
      return copy.sort((a,b)=>{
        const ar=a.cognitive_level==='recall'?0:1,br=b.cognitive_level==='recall'?0:1;
        return ar-br||difficultyRank(a)-difficultyRank(b);
      });
    }
    return copy.sort((a,b)=>difficultyRank(a)-difficultyRank(b));
  }
  function choosePedagogical(d,lessonNo,overrideSeen=null){
    const p=profile(d,lessonNo),m=mutableMeta(d),all=pool(d);
    if(!all.length)return[];
    const target=Math.min(p.target||5,all.length);
    const seen=new Set(overrideSeen||m.seenIds||[]);
    const chosen=[],used=new Set(),conceptUsed=new Set();

    // Consume unseen material first. Repetition is a deliberate fallback only.
    const unseen=all.filter(q=>!seen.has(q.id));
    const seenPool=all.filter(q=>seen.has(q.id));
    const rankList=list=>[...list].sort((a,b)=>score(b,p,seen,conceptUsed)-score(a,p,seen,conceptUsed));
    const unseenRanked=rankList(unseen);
    const seenRanked=rankList(seenPool);
    const unseenPreferred=unseenRanked.filter(q=>p.types.includes(q.activity_type)||p.cog.includes(q.cognitive_level));
    const allPreferred=[...unseenPreferred,...seenRanked.filter(q=>p.types.includes(q.activity_type)||p.cog.includes(q.cognitive_level))];

    function add(q){
      if(!q||used.has(q.id)||chosen.length>=target)return false;
      chosen.push(q);used.add(q.id);if(q.concept_id)conceptUsed.add(q.concept_id);return true;
    }

    const diversityGoal=Math.min(target,p.challenge?4:Math.min(3,new Set(allPreferred.map(q=>q.activity_type)).size));
    for(const type of p.types){
      if(chosen.length>=target||new Set(chosen.map(q=>q.activity_type)).size>=diversityGoal)break;
      add(unseenPreferred.find(x=>x.activity_type===type&&!used.has(x.id)));
    }

    const conceptGoal=Math.min(target,p.challenge?5:3,new Set(all.map(q=>q.concept_id).filter(Boolean)).size);
    for(const q of unseenPreferred){
      if(chosen.length>=target||conceptUsed.size>=conceptGoal)break;
      if(!used.has(q.id)&&q.concept_id&&!conceptUsed.has(q.concept_id))add(q);
    }

    for(const q of unseenPreferred){if(chosen.length>=target)break;add(q)}
    for(const q of unseenRanked){if(chosen.length>=target)break;add(q)}

    if(chosen.length<target){
      const repeatRanked=[...seenRanked].sort((a,b)=>{
        if(p.adaptive){const dw=weakness(b)-weakness(a);if(dw)return dw}
        return score(b,p,seen,conceptUsed)-score(a,p,seen,conceptUsed);
      });
      for(const q of repeatRanked){if(chosen.length>=target)break;add(q)}
    }

    return orderQuestions(chosen,p).slice(0,target);
  }
